   196→   196→  /** Нормализация плана LLM: цепочки + узел-развилка с ветками вариантов */
   197→   197→  private sanitizePlan(json: any): GraphNode[] | null {
   198→   198→    const rawNodes = Array.isArray(json?.nodes) ? json.nodes.slice(0, 12) : null
   199→   199→    if (!rawNodes || rawNodes.length === 0) return null
   200→   200→
   201→   201→    const used = new Set<string>()
   202→   202→    const cleanId = (v: any, fallback: string): string => {
   203→   203→      let s = String(v ?? '').trim().toLowerCase().replace(/[^a-z0-9]/g, '').slice(0, 12)
   204→   204→      if (!s) s = fallback
   205→   205→      while (used.has(s)) s = (s.slice(0, 9) || 'n') + 'x' + Math.floor(Math.random() * 900 + 100)
   206→   206→      used.add(s)
   207→   207→      return s
   208→   208→    }
   209→   209→
   210→   210→    const declared = new Set<string>()
   211→   211→    const cleanDeps = (raw: any, exclude: Set<string>): string[] => {
   212→   212→      const arr: any[] = Array.isArray(raw?.dependsOn) ? raw.dependsOn : []
   213→   213→      const deps = arr
   214→   214→        .map((d: any) => String(d ?? '').trim().toLowerCase().replace(/[^a-z0-9]/g, ''))
   215→   215→        .filter((d: string) => d && declared.has(d) && !exclude.has(d))
   216→   216→      return [...new Set(deps)].slice(0, 5)
   217→   217→    }
   218→   218→
   219→   219→    const nodes: GraphNode[] = []
   220→   220→    rawNodes.forEach((raw: any, i: number) => {
   221→   221→      const id = cleanId(raw?.id, `n${i + 1}`)
   222→   222→      const decisionRaw = raw?.decision
   223→   223→      const optionsRaw = Array.isArray(decisionRaw?.options) ? decisionRaw.options.filter((o: any) => o).slice(0, 3) : []
   224→   224→      const hasDecision = optionsRaw.length >= 2 && !!decisionRaw?.question
   225→   225→
   226→   226→      if (hasDecision) {
   227→   227→        const branchIds = new Set<string>()
   228→   228→        const options: DecisionOption[] = []
   229→   229→        optionsRaw.forEach((ro: any, j: number) => {
   230→   230→          const optId = ['a', 'b', 'c'][j]
   231→   231→          const stepsRaw = Array.isArray(ro?.steps) ? ro.steps.filter((s: any) => s).slice(0, 3) : []
   232→   232→          options.push({
   233→   233→            id: optId,
   234→   234→            label: String(ro?.label || `Вариант ${optId}`).slice(0, 60),
   235→   235→            rationale: String(ro?.rationale || '').slice(0, 320),
   236→   236→            nextSteps: stepsRaw.map((s: any) => String(s?.title || 'шаг ветки').slice(0, 60)),
   237→   237→          })
   238→   238→          // шаги ветки: цепочка от развилки
   239→   239→          let prev = id
   240→   240→          stepsRaw.forEach((rs: any, k: number) => {
   241→   241→            const bid = cleanId(rs?.id, `${id}${optId}${k + 1}`)
   242→   242→            branchIds.add(bid)
   243→   243→            declared.add(bid)
   244→   244→            nodes.push({
   245→   245→              id: bid,
   246→   246→              title: String(rs?.title || `Шаг ветки ${optId}${k + 1}`).slice(0, 60),
   247→   247→              description: String(rs?.description || '').slice(0, 240),
   248→   248→              status: 'pending',
   249→   249→              dependsOn: [prev],
   250→   250→              dynamic: false,
   251→   251→              runs: 0,
   252→   252→              branchOf: { decisionId: id, optionId: optId },
   253→   253→            })
   254→   254→            prev = bid
   255→   255→          })
   256→   256→        })
   257→   257→        declared.add(id)
   258→   258→        nodes.push({
   259→   259→          id,
   260→   260→          kind: 'decision',
   261→   261→          title: String(raw?.title || 'Решение').slice(0, 60),
   262→   262→          description: String(raw?.description || '').slice(0, 240),
   263→   263→          status: 'pending',
   264→   264→          dependsOn: cleanDeps(raw, branchIds),
   265→   265→          dynamic: false,
   266→   266→          runs: 0,
   267→   267→          decision: {
   268→   268→            question: String(decisionRaw.question).slice(0, 300),
   269→   269→            options,
   270→   270→            status: 'pending',
   271→   271→          },
   272→   272→        })
   273→   273→      } else {
   274→   274→        declared.add(id)
   275→   275→        nodes.push({
   276→   276→          id,
   277→   277→          title: String(raw?.title || `Шаг ${i + 1}`).slice(0, 60),
   278→   278→          description: String(raw?.description || '').slice(0, 240),
   279→   279→          status: 'pending',
   280→   280→          dependsOn: cleanDeps(raw, new Set()),
   281→   281→          dynamic: false,
   282→   282→          runs: 0,
   283→   283→        })
   284→   284→      }
   285→   285→    })
   286→   286→    if (nodes.length < 2) return null
   287→   287→    return nodes
   288→   288→  }

   290→   290→  // ─────────────────────── Цикл выполнения ───────────────────────
   291→   291→
   292→   292→  private async runLoop(rt: Runtime) {
   293→   293→    if (rt.loopActive) return
   294→   294→    rt.loopActive = true
   295→   295→    try {
   296→   296→      while (true) {
   297→   297→        const run = rt.snapshot
   298→   298→        if (rt.stopped) {
   299→   299→          this.finalizeStopped(rt)
   300→   300→          return
   301→   301→        }
   302→   302→        const node = this.nextReadyNode(run)
   303→   303→        if (!node) {
   304→   304→          const errored = run.nodes.find((n) => n.status === 'error')
   305→   305→          if (errored) {
   306→   306→            this.log(rt, 'node-error', `Узел «${errored.title}» завершился ошибкой — выполнение остановлено. Доступно вмешательство: «Повторить узел».`)
   307→   307→            this.pushStatus(rt, 'error', run.progress)
   308→   308→            run.finishedAt = now()
   309→   309→            store.save(run)
   310→   310→            return
   311→   311→          }
   312→   312→          const pending = run.nodes.filter((n) => n.status === 'pending')
   313→   313→          if (pending.length) {
   314→   314→            this.log(rt, 'run', 'Обнаружена циклическая зависимость — оставшиеся узлы пропущены')
   315→   315→            pending.forEach((n) => {
   316→   316→              n.status = 'skipped'
   317→   317→              this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...n } })
   318→   318→            })
   319→   319→          }
   320→   320→          this.finalizeCompleted(rt)
   321→   321→          return
   322→   322→        }
   323→   323→
   324→   324→        await this.waitWhilePaused(rt)
   325→   325→        if (rt.stopped) continue
   326→   326→
   327→   327→        node.input = this.buildInput(run, node)
   328→   328→        node.status = 'running'
   329→   329→        node.startedAt = now()
   330→   330→        node.runs = (node.runs ?? 0) + 1
   331→   331→        node.error = undefined
   332→   332→        this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   333→   333→        this.log(rt, 'node-start', `▶ «${node.title}» — выполнение`)
   334→   334→
   335→   335→        // ─── Узел-развилка: рассуждение → предложение вариантов → выбор → гашение веток ───
   336→   336→        if (node.kind === 'decision' && node.decision) {
   337→   337→          const decided = await this.handleDecisionNode(rt, node)
   338→   338→          if (!decided) continue // остановлен во время решения — цикл завершит очистку
   339→   339→          this.recalcProgress(rt)
   340→   340→          this.pushStatus(rt, rt.paused ? 'paused' : 'running', run.progress)
   341→   341→          store.save(run)
   342→   342→          continue
   343→   343→        }
   344→   344→
   345→   345→        try {
   346→   346→          const output =
   347→   347→            run.mode === 'demo' ? await this.demoExecute(node, rt) : await this.llmExecute(run, node)
   348→   348→          node.status = 'done'
   349→   349→          node.output = output
   350→   350→          node.finishedAt = now()
   351→   351→          node.durationMs = node.finishedAt - (node.startedAt ?? node.finishedAt)
   352→   352→          this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   353→   353→          this.log(rt, 'node-done', `✓ «${node.title}» — завершён за ${((node.durationMs ?? 0) / 1000).toFixed(1)} с`)
   354→   354→        } catch (err: any) {
   355→   355→          if (err instanceof StoppedError) {
   356→   356→            node.status = 'pending'
   357→   357→            node.startedAt = undefined
   358→   358→            node.input = undefined
   359→   359→            this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   360→   360→            this.finalizeStopped(rt)
   361→   361→            return
   362→   362→          }
   363→   363→          node.status = 'error'
   364→   364→          node.error = err?.message || String(err)
   365→   365→          node.finishedAt = now()
   366→   366→          node.durationMs = node.finishedAt - (node.startedAt ?? node.finishedAt)
   367→   367→          this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   368→   368→          this.log(rt, 'node-error', `✗ «${node.title}» — ошибка: ${trunc(node.error ?? '', 140)}`)
   369→   369→          continue
   370→   370→        }
   371→   371→
   372→   372→        await this.maybeAdapt(rt, node)
   373→   373→        this.recalcProgress(rt)
   374→   374→        this.pushStatus(rt, rt.paused ? 'paused' : 'running', run.progress)
   375→   375→        store.save(run)
   376→   376→      }
   377→   377→    } finally {
   378→   378→      rt.loopActive = false
   379→   379→    }
   380→   380→  }

   530→   530→  /** Жизненный цикл узла-развилки: рассуждение → предложение вариантов → выбор → активация ветки */
   531→   531→  private async handleDecisionNode(rt: Runtime, node: GraphNode): Promise<boolean> {
   532→   532→    const dec = node.decision!
   533→   533→    this.log(rt, 'decision', `⎇ Развилка «${node.title}»: ${dec.question}`)
   534→   534→    try {
   535→   535→      await this.decisionThinkDelay(rt)
   536→   536→    } catch {
   537→   537→      return this.rollbackDecision(rt, node)
   538→   538→    }
   539→   539→    if (rt.stopped) return this.rollbackDecision(rt, node)
   540→   540→
   541→   541→    dec.status = 'offered'
   542→   542→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   543→   543→    this.emitEvent(rt, {
   544→   544→      type: 'decision:offer',
   545→   545→      ts: now(),
   546→   546→      nodeId: node.id,
   547→   547→      question: dec.question,
   548→   548→      options: dec.options,
   549→   549→    })
   550→   550→
   551→   551→    if (rt.decisionMode === 'manual') {
   552→   552→      this.log(
   553→   553→        rt,
   554→   554→        'decision',
   555→   555→        `⎇ Режим «ручной выбор»: агент предлагает ${dec.options.length} варианта с обоснованием и ожидает решения пользователя`
   556→   556→      )
   557→   557→      const choice = await this.waitForDecision(rt, node)
   558→   558→      if (!choice) return this.rollbackDecision(rt, node)
   559→   559→      this.resolveDecision(rt, node, choice.optionId, choice.by, choice.reasoning)
   560→   560→    } else {
   561→   561→      this.log(rt, 'decision', '⎇ Режим «авто»: агент взвешивает варианты и принимает решение самостоятельно')
   562→   562→      const picked = await this.autoDecide(rt, node)
   563→   563→      if (rt.stopped) return this.rollbackDecision(rt, node)
   564→   564→      this.resolveDecision(rt, node, picked.optionId, 'agent', picked.reasoning)
   565→   565→    }
   566→   566→    return true
   567→   567→  }

   627→   627→  /** Применяет выбор: активирует ветку, гасит альтернативы, фиксирует аудит-трейл */
   628→   628→  private resolveDecision(rt: Runtime, node: GraphNode, optionId: string, by: 'agent' | 'user', reasoning?: string) {
   629→   629→    const run = rt.snapshot
   630→   630→    const dec = node.decision!
   631→   631→    const opt = dec.options.find((o) => o.id === optionId) ?? dec.options[0]
   632→   632→    node.status = 'done'
   633→   633→    node.finishedAt = now()
   634→   634→    node.durationMs = node.finishedAt - (node.startedAt ?? node.finishedAt)
   635→   635→    node.decision = { ...dec, status: 'resolved', chosen: opt.id, chosenBy: by, reasoning: reasoning?.trim() || undefined }
   636→   636→    node.output =
   637→   637→      by === 'agent'
   638→   638→        ? `Решение агента: вариант «${opt.label}». ${reasoning || ''}`.trim()
   639→   639→        : `Выбор пользователя: вариант «${opt.label}». ${opt.rationale}`
   640→   640→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   641→   641→
   642→   642→    // мгновенная визуализация последствий: нереализованные ветки гаснут
   643→   643→    let discarded = 0
   644→   644→    for (const n of run.nodes) {
   645→   645→      if (n.branchOf?.decisionId === node.id && n.branchOf.optionId !== opt.id && n.status === 'pending') {
   646→   646→        n.status = 'discarded'
   647→   647→        discarded++
   648→   648→        this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...n } })
   649→   649→      }
   650→   650→    }
   651→   651→
   652→   652→    this.emitEvent(rt, {
   653→   653→      type: 'decision:resolved',
   654→   654→      ts: now(),
   655→   655→      nodeId: node.id,
   656→   656→      optionId: opt.id,
   657→   657→      by,
   658→   658→      reasoning: node.decision.reasoning,
   659→   659→    })
   660→   660→    this.log(
   661→   661→      rt,
   662→   662→      'decision',
   663→   663→      `⎇ Решение принято (${by === 'agent' ? 'агентом' : 'пользователем'}): «${opt.label}» — ветка активирована, узлов в нереализованных ветках: ${discarded}`
   664→   664→    )
   665→   665→  }

   667→   667→  /** Переключение режима решений на ходу; «авто» во время ожидания — агент решает немедленно */
   668→   668→  setDecisionMode(runId: string, mode: DecisionMode): { ok: boolean; error?: string } {
   669→   669→    const rt = this.runs.get(runId)
   670→   670→    if (!rt) return { ok: false, error: 'Запуск не активен или уже выгружен' }
   671→   671→    if (rt.decisionMode === mode) return { ok: true }
   672→   672→    rt.decisionMode = mode
   673→   673→    rt.snapshot.decisionMode = mode
   674→   674→    this.emitEvent(rt, { type: 'decision:mode', ts: now(), mode })
   675→   675→    this.log(
   676→   676→      rt,
   677→   677→      'control',
   678→   678→      `⎇ Режим решений: ${mode === 'auto' ? '«авто» — агент решает сам' : '«ручной выбор» — агент предлагает варианты и ждёт'}`
   679→   679→    )
   680→   680→    store.save(rt.snapshot)
   681→   681→    // если сейчас ожидается выбор, а включили «авто» — агент принимает решение вместо пользователя
   682→   682→    if (mode === 'auto' && rt.pendingDecision) {
   683→   683→      const pd = rt.pendingDecision
   684→   684→      rt.pendingDecision = null
   685→   685→      const node = rt.snapshot.nodes.find((n) => n.id === pd.nodeId)
   686→   686→      if (node?.decision) {

   688→   688→        this.autoDecide(rt, node)
   689→   689→          .then((picked) => pd.resolve({ optionId: picked.optionId, by: 'agent', reasoning: picked.reasoning }))
   690→   690→          .catch(() => pd.resolve(null))
   691→   691→      } else {
   692→   692→        pd.resolve(null)
   693→   693→      }
   694→   694→    }
   695→   695→    return { ok: true }
   696→   696→  }

   714→   714→  control(runId: string, action: 'pause' | 'resume' | 'stop') {
   715→   715→    const rt = this.runs.get(runId)
   716→   716→    if (!rt) return
   717→   717→    if (action === 'pause') {
   718→   718→      if (!rt.paused && !rt.stopped && rt.snapshot.status === 'running') {
   719→   719→        rt.paused = true
   720→   720→        this.pushStatus(rt, 'paused', rt.snapshot.progress)
   721→   721→        this.log(rt, 'control', '⏸ Пауза: текущий узел завершится, новые не стартуют')
   722→   722→      }
   723→   723→    } else if (action === 'resume') {
   724→   724→      if (rt.paused) {
   725→   725→        rt.paused = false
   726→   726→        this.log(rt, 'control', '▶ Выполнение возобновлено')
   727→   727→        if (rt.snapshot.status === 'paused') {
   728→   728→          this.pushStatus(rt, 'running', rt.snapshot.progress)
   729→   729→        }
   730→   730→      }
   731→   731→    } else if (action === 'stop') {
   732→   732→      if (rt.stopped) return
   733→   733→      if (['completed', 'error', 'stopped'].includes(rt.snapshot.status)) return
   734→   734→      rt.stopped = true
   735→   735→      rt.paused = false
   736→   736→      // если ждём выбор пользователя на развилке — разблокируем ожидание
   737→   737→      if (rt.pendingDecision) {
   738→   738→        const pd = rt.pendingDecision
   739→   739→        rt.pendingDecision = null
   740→   740→        pd.resolve(null)
   741→   741→      }
   742→   742→      this.log(rt, 'control', '⏹ Остановка запрошена — применится после завершения текущего узла')
   743→   743→    }
   744→   744→  }

   746→   746→  private async waitWhilePaused(rt: Runtime) {
   747→   747→    while (rt.paused && !rt.stopped) await sleep(150)

   750→   750→  private finalizeStopped(rt: Runtime) {
   751→   751→    const run = rt.snapshot
   752→   752→    let skipped = 0
   753→   753→    for (const n of run.nodes) {
   754→   754→      if (n.status === 'pending') {
   755→   755→        n.status = 'skipped'
   756→   756→        skipped++
   757→   757→        this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...n } })
   758→   758→      }
   759→   759→    }
   760→   760→    this.recalcProgress(rt)
   761→   761→    this.pushStatus(rt, 'stopped', run.progress)
   762→   762→    const doneCount = run.nodes.filter((n) => n.status === 'done').length
   763→   763→    this.log(rt, 'run', `Выполнение остановлено пользователем · выполнено узлов: ${doneCount}, пропущено: ${skipped}`)
   764→   764→    run.finishedAt = now()
   765→   765→    store.save(run)
   766→   766→  }

   768→   768→  private finalizeCompleted(rt: Runtime) {
   769→   769→    const run = rt.snapshot
   770→   770→    rt.stopped = false
   771→   771→    run.progress = 100
   772→   772→    this.pushStatus(rt, 'completed', 100)
   773→   773→    run.finishedAt = now()
   774→   774→    const dynamic = run.nodes.filter((n) => n.dynamic).length
   775→   775→    const durS = ((run.finishedAt - run.createdAt) / 1000).toFixed(1)
   776→   776→    const last = [...run.nodes].filter((n) => n.status === 'done').pop()
   777→   777→    const summary = `Готово за ${durS} с: ${run.nodes.length} узлов, ${dynamic} добавлено динамически, версий графа — ${run.version}. Итог: ${trunc(last?.output || '—', 220)}`
   778→   778→    run.summary = summary
   779→   779→    this.emitEvent(rt, { type: 'run:summary', ts: now(), summary })
   780→   780→    this.log(rt, 'run', `✓ Задача выполнена за ${durS} с`)
   781→   781→    store.save(run)
   782→   782→  }

   784→   784→  private recalcProgress(rt: Runtime) {
   785→   785→    const run = rt.snapshot
   786→   786→    // нереализованные ветки не считаются работой: прогресс по активным узлам
   787→   787→    const active = run.nodes.filter((n) => n.status !== 'discarded').length || 1
   788→   788→    const done = run.nodes.filter((n) => n.status === 'done').length
   789→   789→    run.progress = Math.round((done / active) * 100)
   790→   790→  }

   792→   792→  private downstreamOf(run: RunSnapshot, nodeId: string): GraphNode[] {
   793→   793→    const out: GraphNode[] = []
   794→   794→    const queue = [nodeId]
   795→   795→    const seen = new Set<string>()
   796→   796→    while (queue.length) {
   797→   797→      const cur = queue.shift()!
   798→   798→      for (const n of run.nodes) {
   799→   799→        if (n.dependsOn.includes(cur) && !seen.has(n.id)) {
   800→   800→          seen.add(n.id)
   801→   801→          out.push(n)
   802→   802→          queue.push(n.id)
   803→   803→        }
   804→   804→      }
   805→   805→    }
   806→   806→    return out
   807→   807→  }

   809→   809→  overrideNode(runId: string, nodeId: string, output: string): { ok: boolean; error?: string; resetCount?: number } {
   810→   810→    const rt = this.runs.get(runId)
   811→   811→    if (!rt) return { ok: false, error: 'Запуск не активен или уже выгружен' }
   812→   812→    const run = rt.snapshot
   813→   813→    const node = run.nodes.find((n) => n.id === nodeId && n.status === 'done')
   814→   814→    if (!node) return { ok: false, error: 'Узел не найден или ещё не завершён' }
   815→   815→    if (rt.stopped) return { ok: false, error: 'Запуск остановлен — переопределение недоступно' }
   816→   816→    const trimmed = (output || '').trim()
   817→   817→    if (!trimmed) return { ok: false, error: 'Результат не может быть пустым' }
   818→   818→
   819→   819→    node.output = trimmed.slice(0, 4000)
   820→   820→    node.overridden = true
   821→   821→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   822→   822→    this.log(rt, 'intervention', `✎ Вмешательство: результат узла «${node.title}» переопределён пользователем`)
   823→   823→
   824→   824→    // нереализованные ветки не восстанавливаем — выбор на развилке остаётся в силе
   825→   825→    const down = this.downstreamOf(run, nodeId).filter((d) => d.status !== 'discarded')
   826→   826→    for (const d of down) {
   827→   827→      d.status = 'pending'
   828→   828→      d.output = undefined
   829→   829→      d.startedAt = undefined
   830→   830→      d.finishedAt = undefined
   831→   831→      d.durationMs = undefined
   832→   832→      d.error = undefined
   833→   833→      d.overridden = false
   834→   834→      this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...d } })
   835→   835→    }
   836→   836→    if (down.length) {
   837→   837→      this.log(rt, 'intervention', `↓ Пересчёт: ${down.length} зависимых узлов сброшены и будут выполнены заново`)
   838→   838→      this.recalcProgress(rt)
   839→   839→    }
   840→   840→    if (down.length && !rt.loopActive) {
   841→   841→      this.pushStatus(rt, 'running', run.progress)
   842→   842→      this.runLoop(rt).catch(() => {})
   843→   843→    }
   844→   844→    store.save(run)
   845→   845→    return { ok: true, resetCount: down.length }
   846→   846→  }

   848→   848→  retryNode(runId: string, nodeId: string): { ok: boolean; error?: string } {
   849→   849→    const rt = this.runs.get(runId)
   850→   850→    if (!rt) return { ok: false, error: 'Запуск не активен' }
   851→   851→    const node = rt.snapshot.nodes.find((n) => n.id === nodeId)
   852→   852→    if (!node || node.status !== 'error') return { ok: false, error: 'Узел не в состоянии ошибки' }
   853→   853→    rt.stopped = false
   854→   854→    node.status = 'pending'
   855→   855→    node.error = undefined
   856→   856→    node.startedAt = undefined
   857→   857→    node.finishedAt = undefined
   858→   858→    node.durationMs = undefined
   859→   859→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   860→   860→    this.log(rt, 'intervention', `↻ Повтор узла «${node.title}» по запросу пользователя`)
   861→   861→    if (!rt.loopActive) {
   862→   862→      this.pushStatus(rt, 'running', rt.snapshot.progress)
   863→   863→      this.runLoop(rt).catch(() => {})
   864→   864→    }
   865→   865→    store.save(rt.snapshot)
   866→   866→    return { ok: true }
   867→   867→  }

   869→   869→  deleteRun(runId: string) {
   870→   870→    const rt = this.runs.get(runId)
   871→   871→    if (rt) {
   872→   872→      rt.stopped = true
   873→   873→      if (rt.pendingDecision) {
   874→   874→        const pd = rt.pendingDecision
   875→   875→        rt.pendingDecision = null
   876→   876→        pd.resolve(null)
   877→   877→      }
   878→   878→      this.runs.delete(runId)
   879→   879→    }
   880→   880→    store.remove(runId)
   881→   881→    this.emit('history:list', { runs: store.history() })
   882→   882→  }
