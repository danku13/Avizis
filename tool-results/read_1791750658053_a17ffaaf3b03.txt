   193→   193→

   235→   235→  /** Нормализация плана LLM: цепочки + узел-развилка с ветками вариантов + certainty */
   236→   236→  private sanitizePlan(json: any, rt?: Runtime): GraphNode[] | null {
   237→   237→    // стартовая самооценка «понимания задачи» от планировщика
   238→   238→    if (rt) {
   239→   239→      const c = Number(json?.certainty)
   240→   240→      if (Number.isFinite(c)) rt.snapshot.certainty = Math.min(1, Math.max(0, c))
   241→   241→    }
   242→   242→    const rawNodes = Array.isArray(json?.nodes) ? json.nodes.slice(0, 12) : null
   243→   243→    if (!rawNodes || rawNodes.length === 0) return null
   244→   244→
   245→   245→    const used = new Set<string>()
   246→   246→    const cleanId = (v: any, fallback: string): string => {
   247→   247→      let s = String(v ?? '').trim().toLowerCase().replace(/[^a-z0-9]/g, '').slice(0, 12)
   248→   248→      if (!s) s = fallback
   249→   249→      while (used.has(s)) s = (s.slice(0, 9) || 'n') + 'x' + Math.floor(Math.random() * 900 + 100)
   250→   250→      used.add(s)
   251→   251→      return s
   252→   252→    }
   253→   253→
   254→   254→    const declared = new Set<string>()
   255→   255→    const cleanDeps = (raw: any, exclude: Set<string>): string[] => {
   256→   256→      const arr: any[] = Array.isArray(raw?.dependsOn) ? raw.dependsOn : []
   257→   257→      const deps = arr
   258→   258→        .map((d: any) => String(d ?? '').trim().toLowerCase().replace(/[^a-z0-9]/g, ''))
   259→   259→        .filter((d: string) => d && declared.has(d) && !exclude.has(d))
   260→   260→      return [...new Set(deps)].slice(0, 5)
   261→   261→    }
   262→   262→
   263→   263→    const nodes: GraphNode[] = []
   264→   264→    rawNodes.forEach((raw: any, i: number) => {
   265→   265→      const id = cleanId(raw?.id, `n${i + 1}`)
   266→   266→      const decisionRaw = raw?.decision
   267→   267→      const optionsRaw = Array.isArray(decisionRaw?.options) ? decisionRaw.options.filter((o: any) => o).slice(0, 4) : []
   268→   268→      const hasDecision = optionsRaw.length >= 2 && !!decisionRaw?.question
   269→   269→
   270→   270→      if (hasDecision) {
   271→   271→        const branchIds = new Set<string>()
   272→   272→        const selection: 'single' | 'mixed' = decisionRaw?.selection === 'mixed' ? 'mixed' : 'single'
   273→   273→        const domain: DecisionDomain | undefined = DECISION_DOMAINS.includes(decisionRaw?.domain)
   274→   274→          ? decisionRaw.domain
   275→   275→          : undefined
   276→   276→        const certaintyAfter = Number(decisionRaw?.certaintyAfter)
   277→   277→        const options: DecisionOption[] = []
   278→   278→        optionsRaw.forEach((ro: any, j: number) => {
   279→   279→          const optId = ['a', 'b', 'c', 'd'][j]
   280→   280→          const stepsRaw = Array.isArray(ro?.steps) ? ro.steps.filter((s: any) => s).slice(0, 3) : []
   281→   281→          options.push({
   282→   282→            id: optId,
   283→   283→            label: String(ro?.label || `Вариант ${optId}`).slice(0, 60),
   284→   284→            rationale: String(ro?.rationale || '').slice(0, 800),
   285→   285→            nextSteps: stepsRaw.map((s: any) => String(s?.title || 'шаг ветки').slice(0, 60)),
   286→   286→          })
   287→   287→          // шаги ветки: цепочка от развилки
   288→   288→          let prev = id
   289→   289→          stepsRaw.forEach((rs: any, k: number) => {
   290→   290→            const bid = cleanId(rs?.id, `${id}${optId}${k + 1}`)
   291→   291→            branchIds.add(bid)
   292→   292→            declared.add(bid)
   293→   293→            nodes.push({
   294→   294→              id: bid,
   295→   295→              title: String(rs?.title || `Шаг ветки ${optId}${k + 1}`).slice(0, 60),
   296→   296→              description: String(rs?.description || '').slice(0, 800),
   297→   297→              status: 'pending',
   298→   298→              dependsOn: [prev],
   299→   299→              dynamic: false,
   300→   300→              runs: 0,
   301→   301→              branchOf: { decisionId: id, optionId: optId },
   302→   302→            })
   303→   303→            prev = bid
   304→   304→          })
   305→   305→        })
   306→   306→        declared.add(id)
   307→   307→        nodes.push({
   308→   308→          id,
   309→   309→          kind: 'decision',
   310→   310→          title: String(raw?.title || 'Решение').slice(0, 60),
   311→   311→          description: String(raw?.description || '').slice(0, 800),
   312→   312→          status: 'pending',
   313→   313→          dependsOn: cleanDeps(raw, branchIds),
   314→   314→          dynamic: false,
   315→   315→          runs: 0,
   316→   316→          decision: {
   317→   317→            question: String(decisionRaw.question).slice(0, 800),
   318→   318→            domain,
   319→   319→            selection,
   320→   320→            certaintyAfter: Number.isFinite(certaintyAfter) ? Math.min(1, Math.max(0, certaintyAfter)) : undefined,
   321→   321→            options,
   322→   322→            status: 'pending',
   323→   323→          },
   324→   324→        })
   325→   325→      } else {
   326→   326→        declared.add(id)
   327→   327→        nodes.push({
   328→   328→          id,
   329→   329→          title: String(raw?.title || `Шаг ${i + 1}`).slice(0, 60),
   330→   330→          description: String(raw?.description || '').slice(0, 800),
   331→   331→          status: 'pending',
   332→   332→          dependsOn: cleanDeps(raw, new Set()),
   333→   333→          dynamic: false,
   334→   334→          runs: 0,
   335→   335→        })
   336→   336→      }
   337→   337→    })
   338→   338→    if (nodes.length < 2) return null
   339→   339→    return nodes
   340→   340→  }

   342→   342→  // ─────────────────────── Цикл выполнения ───────────────────────

   344→   344→  private async runLoop(rt: Runtime) {
   345→   345→    if (rt.loopActive) return
   346→   346→    rt.loopActive = true
   347→   347→    try {
   348→   348→      while (true) {
   349→   349→        const run = rt.snapshot
   350→   350→        if (rt.stopped) {
   351→   351→          this.finalizeStopped(rt)
   352→   352→          return
   353→   353→        }
   354→   354→        const node = this.nextReadyNode(run)
   355→   355→        if (!node) {
   356→   356→          const errored = run.nodes.find((n) => n.status === 'error')
   357→   357→          if (errored) {
   358→   358→            this.log(rt, 'node-error', `Узел «${errored.title}» завершился ошибкой — выполнение остановлено. Доступно вмешательство: «Повторить узел».`)
   359→   359→            this.pushStatus(rt, 'error', run.progress)
   360→   360→            run.finishedAt = now()
   361→   361→            store.save(run)
   362→   362→            return
   363→   363→          }
   364→   364→          const pending = run.nodes.filter((n) => n.status === 'pending')
   365→   365→          if (pending.length) {
   366→   366→            this.log(rt, 'run', 'Обнаружена циклическая зависимость — оставшиеся узлы пропущены')
   367→   367→            pending.forEach((n) => {
   368→   368→              n.status = 'skipped'
   369→   369→              this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...n } })
   370→   370→            })
   371→   371→          }
   372→   372→          this.finalizeCompleted(rt)
   373→   373→          return
   374→   374→        }
   375→   375→
   376→   376→        await this.waitWhilePaused(rt)
   377→   377→        if (rt.stopped) continue
   378→   378→
   379→   379→        node.input = this.buildInput(run, node)
   380→   380→        node.status = 'running'
   381→   381→        node.startedAt = now()
   382→   382→        node.runs = (node.runs ?? 0) + 1
   383→   383→        node.error = undefined
   384→   384→        this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   385→   385→        this.log(rt, 'node-start', `▶ «${node.title}» — выполнение`)
   386→   386→
   387→   387→        // ─── Узел-развилка: рассуждение → предложение вариантов → выбор → гашение веток ───
   388→   388→        if (node.kind === 'decision' && node.decision) {
   389→   389→          const decided = await this.handleDecisionNode(rt, node)
   390→   390→          if (!decided) continue // остановлен во время решения — цикл завершит очистку
   391→   391→          this.recalcProgress(rt)
   392→   392→          this.pushStatus(rt, rt.paused ? 'paused' : 'running', run.progress)
   393→   393→          store.save(run)
   394→   394→          continue
   395→   395→        }
   396→   396→
   397→   397→        try {
   398→   398→          const output =
   399→   399→            run.mode === 'demo' ? await this.demoExecute(node, rt) : await this.llmExecute(run, node)
   400→   400→          node.status = 'done'
   401→   401→          node.output = output
   402→   402→          node.finishedAt = now()
   403→   403→          node.durationMs = node.finishedAt - (node.startedAt ?? node.finishedAt)
   404→   404→          this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   405→   405→          this.log(rt, 'node-done', `✓ «${node.title}» — завершён за ${((node.durationMs ?? 0) / 1000).toFixed(1)} с`)
   406→   406→        } catch (err: any) {
   407→   407→          if (err instanceof StoppedError) {
   408→   408→            node.status = 'pending'
   409→   409→            node.startedAt = undefined
   410→   410→            node.input = undefined
   411→   411→            this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   412→   412→            this.finalizeStopped(rt)
   413→   413→            return
   414→   414→          }
   415→   415→          node.status = 'error'
   416→   416→          node.error = err?.message || String(err)
   417→   417→          node.finishedAt = now()
   418→   418→          node.durationMs = node.finishedAt - (node.startedAt ?? node.finishedAt)
   419→   419→          this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   420→   420→          this.log(rt, 'node-error', `✗ «${node.title}» — ошибка: ${trunc(node.error ?? '', 140)}`)
   421→   421→          continue
   422→   422→        }
   423→   423→
   424→   424→        await this.maybeAdapt(rt, node)
   425→   425→        this.recalcProgress(rt)
   426→   426→        this.pushStatus(rt, rt.paused ? 'paused' : 'running', run.progress)
   427→   427→        store.save(run)
   428→   428→      }
   429→   429→    } finally {
   430→   430→      rt.loopActive = false
   431→   431→    }
   432→   432→  }

   434→   434→  private nextReadyNode(run: RunSnapshot): GraphNode | undefined {
   435→   435→    const byId = new Map(run.nodes.map((n) => [n.id, n]))
   436→   436→    // «discarded» = ветка не реализована: зависимость от неё считается удовлетворённой
   437→   437→    const ready = (n: GraphNode) =>
   438→   438→      n.dependsOn.every((d) => {
   439→   439→        const dep = byId.get(d)
   440→   440→        return !dep || dep.status === 'done' || dep.status === 'skipped' || dep.status === 'discarded'
   441→   441→      })
   442→   442→    return run.nodes.find((n) => n.status === 'pending' && ready(n))
   443→   443→  }

   445→   445→  private buildInput(run: RunSnapshot, node: GraphNode): string {
   446→   446→    const deps = node.dependsOn
   447→   447→      .map((id) => run.nodes.find((n) => n.id === id))
   448→   448→      .filter((n): n is GraphNode => !!n && n.status === 'done')
   449→   449→    // Полные тексты без обрезки: node.input — это данные инспектора (Вход),
   450→   450→    // а не только промпт. Обрезка промптов — отдельно в llmExecute/decider.
   451→   451→    if (!deps.length) return `Задача: ${run.task}`
   452→   452→    return `Задача: ${run.task}\n\nРезультаты предыдущих шагов:\n${deps
   453→   453→      .map((d) => `• «${d.title}»: ${d.output || '—'}`)
   454→   454→      .join('\n')}`
   455→   455→  }

   626→   626→  private insertNode(rt: Runtime, spec: { id: string; title: string; description: string; dependsOn: string[] }, reason: string) {
   627→   627→    const run = rt.snapshot
   628→   628→    const newNode: GraphNode = {
   629→   629→      id: spec.id,
   630→   630→      title: spec.title,
   631→   631→      description: spec.description,
   632→   632→      status: 'pending',
   633→   633→      dependsOn: [...spec.dependsOn],
   634→   634→      dynamic: true,
   635→   635→      reason,
   636→   636→      addedAtVersion: run.version + 1,
   637→   637→      runs: 0,
   638→   638→    }
   639→   639→    const changed: GraphNode[] = [newNode]
   640→   640→    const anchor = spec.dependsOn[0]
   641→   641→    // Перепривязываем ещё не начатые узлы, зависящие от якоря, на новый узел
   642→   642→    for (const n of run.nodes) {
   643→   643→      if (n.status === 'pending' && n.dependsOn.includes(anchor)) {
   644→   644→        n.dependsOn = [...new Set(n.dependsOn.map((d) => (d === anchor ? newNode.id : d)))]
   645→   645→        changed.push({ ...n })
   646→   646→      }
   647→   647→    }
   648→   648→    run.nodes.push(newNode)
   649→   649→    run.version += 1
   650→   650→    rt.dynamicAdded += 1
   651→   651→    run.versions.push({ version: run.version, createdAt: now(), description: reason, nodeIds: run.nodes.map((n) => n.id) })
   652→   652→    this.emitEvent(rt, { type: 'graph:adapt', ts: now(), reason, version: run.version, nodes: changed.map((n) => ({ ...n })) })
   653→   653→    this.log(rt, 'adapt', `＋ Граф адаптирован (v${run.version}): добавлен узел «${newNode.title}». Причина: ${reason}`)
   654→   654→    store.save(run)
   655→   655→  }

   657→   657→  /**
   658→   658→   * Вставка эмерджентной развилки: агент проанализировал результаты шага и
   659→   659→   * обнаружил неоднозначность, влияющую на итог. Создаёт узел-вопрос с ветками
   660→   660→   * вариантов; ещё не начатые узлы, ждавшие якорь, переводятся на хвосты веток —
   661→   661→   * итог корректно дождётся только реализованных веток.
   662→   662→   */
   663→   663→  private insertDecision(
   664→   664→    rt: Runtime,
   665→   665→    anchorId: string,
   666→   666→    spec: {
   667→   667→      id: string
   668→   668→      title: string
   669→   669→      description: string
   670→   670→      question: string
   671→   671→      domain?: string
   672→   672→      selection: 'single' | 'mixed'
   673→   673→      certaintyAfter?: number
   674→   674→      options: Array<{ id: string; label: string; rationale: string; steps: Array<{ id?: string; title: string; description: string }> }>
   675→   675→    },
   676→   676→    reason: string
   677→   677→  ) {
   678→   678→    const run = rt.snapshot
   679→   679→    const domain: DecisionDomain | undefined = DECISION_DOMAINS.includes(spec.domain as DecisionDomain)
   680→   680→      ? (spec.domain as DecisionDomain)
   681→   681→      : undefined
   682→   682→    const version = run.version + 1
   683→   683→
   684→   684→    const decNode: GraphNode = {
   685→   685→      id: spec.id,
   686→   686→      kind: 'decision',
   687→   687→      title: spec.title,
   688→   688→      description: spec.description,
   689→   689→      status: 'pending',
   690→   690→      dependsOn: [anchorId],
   691→   691→      dynamic: true,
   692→   692→      reason,
   693→   693→      addedAtVersion: version,
   694→   694→      runs: 0,
   695→   695→      decision: {
   696→   696→        question: spec.question,
   697→   697→        domain,
   698→   698→        selection: spec.selection,
   699→   699→        certaintyAfter: spec.certaintyAfter,
   700→   700→        options: spec.options.map((o) => ({
   701→   701→          id: o.id,
   702→   702→          label: o.label,
   703→   703→          rationale: o.rationale,
   704→   704→          nextSteps: o.steps.map((s) => s.title),
   705→   705→        })),
   706→   706→        status: 'pending',
   707→   707→      },
   708→   708→    }
   709→   709→    const changed: GraphNode[] = [decNode]
   710→   710→    const tails: string[] = []
   711→   711→
   712→   712→    // ветки вариантов: цепочки от развилки
   713→   713→    spec.options.forEach((o) => {
   714→   714→      let prev = spec.id
   715→   715→      o.steps.forEach((s, k) => {
   716→   716→        const bid = s.id || `${spec.id}${o.id}${k + 1}`
   717→   717→        const branchNode: GraphNode = {
   718→   718→          id: bid,
   719→   719→          title: s.title,
   720→   720→          description: s.description,
   721→   721→          status: 'pending',
   722→   722→          dependsOn: [prev],
   723→   723→          dynamic: true,
   724→   724→          reason,
   725→   725→          addedAtVersion: version,
   726→   726→          runs: 0,
   727→   727→          branchOf: { decisionId: spec.id, optionId: o.id },
   728→   728→        }
   729→   729→        run.nodes.push(branchNode)
   730→   730→        changed.push({ ...branchNode })
   731→   731→        prev = bid
   732→   732→      })
   733→   733→      if (prev === spec.id) return // у варианта нет шагов — ветка «прямое продолжение»
   734→   734→      tails.push(prev)
   735→   735→    })
   736→   736→    run.nodes.push(decNode)
   737→   737→
   738→   738→    // перепривязка: ещё не начатые узлы, ждавшие якорь, ждут хвосты ВСЕХ веток
   739→   739→    // (невыбранные ветки станут discarded — зависимость удовлетворится автоматически)
   740→   740→    for (const n of run.nodes) {
   741→   741→      if (n.id !== decNode.id && n.status === 'pending' && n.dependsOn.includes(anchorId)) {
   742→   742→        n.dependsOn = [...new Set(n.dependsOn.flatMap((d) => (d === anchorId ? (tails.length ? tails : [anchorId]) : [d])))].slice(0, 12)
   743→   743→        changed.push({ ...n })
   744→   744→      }
   745→   745→    }
   746→   746→
   747→   747→    run.version = version
   748→   748→    rt.dynamicAdded += 1
   749→   749→    rt.emergentAdded += 1
   750→   750→    run.versions.push({ version, createdAt: now(), description: reason, nodeIds: run.nodes.map((n) => n.id) })
   751→   751→    this.emitEvent(rt, { type: 'graph:adapt', ts: now(), reason, version, nodes: changed.map((n) => ({ ...n })) })
   752→   752→    this.log(
   753→   753→      rt,
   754→   754→      'adapt',
   755→   755→      `＋ Эмерджентная развилка (v${version}): «${spec.title}» — ${spec.options.length} варианта${
   756→   756→        spec.selection === 'mixed' ? ' (смешанный выбор допустим)' : ''
   757→   757→      }. Причина: ${reason}`
   758→   758→    )
   759→   759→    store.save(run)
   760→   760→  }

   762→   762→  /** Публикация самооценки «понимания задачи»: насколько узкий вектор решения */
   763→   763→  private emitCertainty(rt: Runtime, certainty: number, note?: string) {
   764→   764→    const c = Math.min(1, Math.max(0, Number.isFinite(certainty) ? certainty : 0))
   765→   765→    rt.snapshot.certainty = c
   766→   766→    this.emitEvent(rt, { type: 'run:certainty', ts: now(), certainty: c, note })
   767→   767→    const pct = Math.round(c * 100)
   768→   768→    this.log(rt, 'decision', `◎ Понимание задачи: ${pct}% — ${note || 'агент переоценил вектор решения'}`)
   769→   769→  }

   773→   773→  /** Жизненный цикл узла-развилки: рассуждение → предложение вариантов → выбор → активация ветки */
   774→   774→  private async handleDecisionNode(rt: Runtime, node: GraphNode): Promise<boolean> {
   775→   775→    const dec = node.decision!
   776→   776→    this.log(rt, 'decision', `⎇ Развилка «${node.title}»: ${dec.question}`)
   777→   777→    try {
   778→   778→      await this.decisionThinkDelay(rt)
   779→   779→    } catch {
   780→   780→      return this.rollbackDecision(rt, node)
   781→   781→    }
   782→   782→    if (rt.stopped) return this.rollbackDecision(rt, node)
   783→   783→
   784→   784→    dec.status = 'offered'
   785→   785→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   786→   786→    this.emitEvent(rt, {
   787→   787→      type: 'decision:offer',
   788→   788→      ts: now(),
   789→   789→      nodeId: node.id,
   790→   790→      question: dec.question,
   791→   791→      options: dec.options,
   792→   792→    })
   793→   793→
   794→   794→    if (rt.decisionMode === 'manual') {
   795→   795→      this.log(
   796→   796→        rt,
   797→   797→        'decision',
   798→   798→        `⎇ Режим «ручной выбор»: агент предлагает ${dec.options.length} варианта с обоснованием и ожидает решения пользователя${
   799→   799→          dec.selection === 'mixed' ? ' (допустимо смешанное решение из нескольких)' : ''
   800→   800→        }`
   801→   801→      )
   802→   802→      const choice = await this.waitForDecision(rt, node)
   803→   803→      if (!choice) return this.rollbackDecision(rt, node)
   804→   804→      this.resolveDecision(rt, node, choice.optionIds, choice.by, choice.reasoning)
   805→   805→    } else {
   806→   806→      this.log(rt, 'decision', '⎇ Режим «авто»: агент взвешивает варианты и принимает решение самостоятельно')
   807→   807→      const picked = await this.autoDecide(rt, node)
   808→   808→      if (rt.stopped) return this.rollbackDecision(rt, node)
   809→   809→      this.resolveDecision(rt, node, picked.optionIds, 'agent', picked.reasoning)
   810→   810→    }
   811→   811→    return true
   812→   812→  }

   814→   814→  /** Откат развилки (остановка во время решения): узел возвращается в очередь */
   815→   815→  private rollbackDecision(rt: Runtime, node: GraphNode): boolean {
   816→   816→    node.status = 'pending'
   817→   817→    node.startedAt = undefined
   818→   818→    node.input = undefined
   819→   819→    if (node.decision) node.decision = { ...node.decision, status: 'pending' }
   820→   820→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   821→   821→    return false
   822→   822→  }

   824→   824→  /** Фаза рассуждения на развилке: пауза с уважением stop/pause */
   825→   825→  private async decisionThinkDelay(rt: Runtime): Promise<void> {
   826→   826→    const deadline = now() + (rt.snapshot.mode === 'llm' ? 700 : 1600)
   827→   827→    while (now() < deadline) {
   828→   828→      if (rt.stopped) throw new StoppedError()
   829→   829→      while (rt.paused && !rt.stopped) await sleep(150)
   830→   830→      if (rt.stopped) throw new StoppedError()
   831→   831→      await sleep(120)
   832→   832→    }
   833→   833→  }

   842→   842→  /** Авто-режим: демо — скриптованный выбор; LLM — вызов модуля принятия решений */
   843→   843→  private async autoDecide(rt: Runtime, node: GraphNode): Promise<{ optionIds: string[]; reasoning: string }> {
   844→   844→    const dec = node.decision!
   845→   845→    if (rt.snapshot.mode === 'demo') {
   846→   846→      await sleep(1200)
   847→   847→      const scripted = DEMO_WRITING_AUTO_PICKS[node.id]
   848→   848→      if (scripted) return { optionIds: [...scripted.optionIds], reasoning: scripted.reasoning }
   849→   849→      return { optionIds: [...DEMO_AUTO_PICK.optionIds], reasoning: DEMO_AUTO_PICK.reasoning }
   850→   850→    }
   851→   851→    const mixed = dec.selection === 'mixed'
   852→   852→    const user = `Задача: «${rt.snapshot.task}»
   853→   853→
   854→   854→Развилка: ${dec.question}
   855→   855→Тип выбора: ${mixed ? 'mixed — можно выбрать НЕСКОЛЬКО вариантов (смешанное решение)' : 'single — нужно выбрать ровно один вариант'}
   856→   856→
   857→   857→Варианты:
   858→   858→${dec.options.map((o) => `- id «${o.id}» — ${o.label}: ${o.rationale}`).join('\n')}
   859→   859→
   860→   860→${node.input || ''}
   861→   861→
   862→   862→${mixed ? 'Выбери один или несколько вариантов (id из списка) и обоснуй состав.' : 'Выбери один вариант (id из списка) и обоснуй выбор.'} Ответ строго JSON:
   863→   863→${mixed ? '{"optionIds":["<id1>","<id2>"],"reasoning":"обоснование в 1–2 предложениях"}' : '{"optionIds":["<id>"],"reasoning":"обоснование в 1–2 предложениях"}'}`
   864→   864→    try {
   865→   865→      const json = await chatJson(DECIDER_SYS, user, 45_000, 1)

   867→   867→        ? json.optionIds
   868→   868→        : Array.isArray(json?.options)
   869→   869→          ? json.options
   870→   870→          : json?.optionId
   871→   871→            ? [json.optionId]
   872→   872→            : []
   873→   873→      const valid = rawIds
   874→   874→        .map((v) => String(v ?? '').trim().toLowerCase())
   875→   875→        .filter((v) => dec.options.some((o) => o.id === v))

   878→   878→        const picked = mixed ? unique : unique.slice(0, 1)

   880→   880→      }
   881→   881→    } catch {
   882→   882→      /* fallback ниже */
   883→   883→    }
   884→   884→    return { optionIds: [dec.options[0].id], reasoning: 'Обоснование LLM недоступно — выбран первый вариант по умолчанию' }
   885→   885→  }

   887→   887→  /** Применяет выбор (один или несколько вариантов): активирует ветки, гасит альтернативы, фиксирует аудит-трейл */
   888→   888→  private resolveDecision(rt: Runtime, node: GraphNode, optionIds: string[], by: 'agent' | 'user', reasoning?: string) {
   889→   889→    const run = rt.snapshot
   890→   890→    const dec = node.decision!
   891→   891→    const single = dec.selection !== 'mixed'
   892→   892→    // валидация: только известные id; single — ровно один
   893→   893→    let picked = optionIds.map((id) => id.trim().toLowerCase()).filter((id) => dec.options.some((o) => o.id === id))
   894→   894→    picked = [...new Set(picked)]
   895→   895→    if (!picked.length) picked = [dec.options[0].id]
   896→   896→    if (single) picked = picked.slice(0, 1)
   897→   897→    const opts = picked.map((id) => dec.options.find((o) => o.id === id)!)
   898→   898→    const isMixedResult = picked.length > 1
   899→   899→    const pickedLabel = opts.map((o) => `«${o.label}»`).join(isMixedResult ? ' + ' : '')
   900→   900→
   901→   901→    node.status = 'done'
   902→   902→    node.finishedAt = now()
   903→   903→    node.durationMs = node.finishedAt - (node.startedAt ?? node.finishedAt)
   904→   904→    node.decision = { ...dec, status: 'resolved', chosen: picked, chosenBy: by, reasoning: reasoning?.trim() || undefined }
   905→   905→    node.output =
   906→   906→      by === 'agent'
   907→   907→        ? `${isMixedResult ? `Смешанное решение агента: ${pickedLabel}.` : `Решение агента: вариант ${pickedLabel}.`} ${reasoning || ''}`.trim()
   908→   908→        : `${isMixedResult ? `Смешанный выбор пользователя: ${pickedLabel}.` : `Выбор пользователя: вариант ${pickedLabel}.`} ${opts.map((o) => o.rationale).join(' ')}`
   909→   909→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   910→   910→
   911→   911→    // мгновенная визуализация последствий: нереализованные ветки гаснут
   912→   912→    let discarded = 0
   913→   913→    for (const n of run.nodes) {
   914→   914→      if (n.branchOf?.decisionId === node.id && !picked.includes(n.branchOf.optionId) && n.status === 'pending') {
   915→   915→        n.status = 'discarded'
   916→   916→        discarded++
   917→   917→        this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...n } })
   918→   918→      }
   919→   919→    }
   920→   920→
   921→   921→    this.emitEvent(rt, {
   922→   922→      type: 'decision:resolved',
   923→   923→      ts: now(),
   924→   924→      nodeId: node.id,
   925→   925→      optionIds: picked,
   926→   926→      by,
   927→   927→      reasoning: node.decision.reasoning,
   928→   928→    })
   929→   929→    this.log(
   930→   930→      rt,
   931→   931→      'decision',
   932→   932→      `⎇ Решение принято (${by === 'agent' ? 'агентом' : 'пользователем'}): ${pickedLabel}${
   933→   933→        isMixedResult ? ' — ветки активированы' : ' — ветка активирована'
   934→   934→      }, узлов в нереализованных ветках: ${discarded}`
   935→   935→    )
   936→   936→
   937→   937→    // закрытие вопроса сужает вектор понимания задачи
   938→   938→    const after =
   939→   939→      dec.certaintyAfter ?? Math.min(0.95, (rt.snapshot.certainty ?? 0.4) + 0.15)
   940→   940→    this.emitCertainty(
   941→   941→      rt,
   942→   942→      Math.max(rt.snapshot.certainty ?? 0, after),
   943→   943→      isMixedResult
   944→   944→        ? `смешанное решение по вопросу «${trunc(node.title, 48)}»`
   945→   945→        : `вопрос «${trunc(node.title, 48)}» закрыт`
   946→   946→    )
   947→   947→  }

   949→   949→  /** Переключение режима решений на ходу; «авто» во время ожидания — агент решает немедленно */
   950→   950→  setDecisionMode(runId: string, mode: DecisionMode): { ok: boolean; error?: string } {
   951→   951→    const rt = this.runs.get(runId)
   952→   952→    if (!rt) return { ok: false, error: 'Запуск не активен или уже выгружен' }
   953→   953→    if (rt.decisionMode === mode) return { ok: true }
   954→   954→    rt.decisionMode = mode
   955→   955→    rt.snapshot.decisionMode = mode
   956→   956→    this.emitEvent(rt, { type: 'decision:mode', ts: now(), mode })
   957→   957→    this.log(
   958→   958→      rt,
   959→   959→      'control',
   960→   960→      `⎇ Режим решений: ${mode === 'auto' ? '«авто» — агент решает сам' : '«ручной выбор» — агент предлагает варианты и ждёт'}`
   961→   961→    )
   962→   962→    store.save(rt.snapshot)
   963→   963→    // если сейчас ожидается выбор, а включили «авто» — агент принимает решение вместо пользователя
   964→   964→    if (mode === 'auto' && rt.pendingDecision) {
   965→   965→      const pd = rt.pendingDecision
   966→   966→      rt.pendingDecision = null
   967→   967→      const node = rt.snapshot.nodes.find((n) => n.id === pd.nodeId)
   968→   968→      if (node?.decision) {
   969→   969→        this.log(rt, 'decision', '⎇ Ожидание выбора прервано: агент принимает решение вместо пользователя')
   970→   970→        this.autoDecide(rt, node)
   971→   971→          .then((picked) => pd.resolve({ optionIds: picked.optionIds, by: 'agent', reasoning: picked.reasoning }))
   972→   972→          .catch(() => pd.resolve(null))
   973→   973→      } else {
   974→   974→        pd.resolve(null)
   975→   975→      }
   976→   976→    }
   977→   977→    return { ok: true }
   978→   978→  }

   980→   980→  /** Выбор пользователя на развилке (из модалки): один или несколько вариантов */
   981→   981→  resolveDecisionFromUser(runId: string, nodeId: string, optionIds: string[]): { ok: boolean; error?: string } {
   982→   982→    const rt = this.runs.get(runId)
   983→   983→    if (!rt) return { ok: false, error: 'Запуск не активен' }
   984→   984→    const pd = rt.pendingDecision
   985→   985→    if (!pd || pd.nodeId !== nodeId) return { ok: false, error: 'Решение не ожидается (уже принято или запуск завершён)' }
   986→   986→    const node = rt.snapshot.nodes.find((n) => n.id === nodeId)
   987→   987→    const dec = node?.decision
   988→   988→    const ids = (optionIds || []).map((id) => String(id).trim().toLowerCase())
   989→   989→    if (!dec || !ids.length || !ids.every((id) => dec.options.some((o) => o.id === id))) {
   990→   990→      return { ok: false, error: 'Неизвестный вариант' }
   991→   991→    }
   992→   992→    rt.pendingDecision = null
   993→   993→    pd.resolve({ optionIds: ids, by: 'user' })
   994→   994→    return { ok: true }
   995→   995→  }

   999→   999→  control(runId: string, action: 'pause' | 'resume' | 'stop') {
  1000→  1000→    const rt = this.runs.get(runId)
  1001→  1001→    if (!rt) return
  1002→  1002→    if (action === 'pause') {
  1003→  1003→      if (!rt.paused && !rt.stopped && rt.snapshot.status === 'running') {
  1004→  1004→        rt.paused = true
  1005→  1005→        this.pushStatus(rt, 'paused', rt.snapshot.progress)
  1006→  1006→        this.log(rt, 'control', '⏸ Пауза: текущий узел завершится, новые не стартуют')
  1007→  1007→      }
  1008→  1008→    } else if (action === 'resume') {
  1009→  1009→      if (rt.paused) {
  1010→  1010→        rt.paused = false
  1011→  1011→        this.log(rt, 'control', '▶ Выполнение возобновлено')
  1012→  1012→        if (rt.snapshot.status === 'paused') {
  1013→  1013→          this.pushStatus(rt, 'running', rt.snapshot.progress)
  1014→  1014→        }
  1015→  1015→      }
  1016→  1016→    } else if (action === 'stop') {
  1017→  1017→      if (rt.stopped) return
  1018→  1018→      if (['completed', 'error', 'stopped'].includes(rt.snapshot.status)) return
  1019→  1019→      rt.stopped = true
  1020→  1020→      rt.paused = false
  1021→  1021→      // если ждём выбор пользователя на развилке — разблокируем ожидание
  1022→  1022→      if (rt.pendingDecision) {
  1023→  1023→        const pd = rt.pendingDecision
  1024→  1024→        rt.pendingDecision = null
  1025→  1025→        pd.resolve(null)
  1026→  1026→      }
  1027→  1027→      this.log(rt, 'control', '⏹ Остановка запрошена — применится после завершения текущего узла')
  1028→  1028→    }
  1029→  1029→  }

  1035→  1035→  private finalizeStopped(rt: Runtime) {
  1036→  1036→    const run = rt.snapshot
  1037→  1037→    let skipped = 0
  1038→  1038→    for (const n of run.nodes) {
  1039→  1039→      if (n.status === 'pending') {
  1040→  1040→        n.status = 'skipped'
  1041→  1041→        skipped++
  1042→  1042→        this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...n } })
  1043→  1043→      }
  1044→  1044→    }
  1045→  1045→    this.recalcProgress(rt)
  1046→  1046→    this.pushStatus(rt, 'stopped', run.progress)
  1047→  1047→    const doneCount = run.nodes.filter((n) => n.status === 'done').length
  1048→  1048→    this.log(rt, 'run', `Выполнение остановлено пользователем · выполнено узлов: ${doneCount}, пропущено: ${skipped}`)
  1049→  1049→    run.finishedAt = now()
  1050→  1050→    store.save(run)
  1051→  1051→  }

  1053→  1053→  private finalizeCompleted(rt: Runtime) {
  1054→  1054→    const run = rt.snapshot
  1055→  1055→    rt.stopped = false
  1056→  1056→    run.progress = 100
  1057→  1057→    this.pushStatus(rt, 'completed', 100)
  1058→  1058→    run.finishedAt = now()
  1059→  1059→    const dynamic = run.nodes.filter((n) => n.dynamic).length
  1060→  1060→    const durS = ((run.finishedAt - run.createdAt) / 1000).toFixed(1)
  1061→  1061→    const last = [...run.nodes].filter((n) => n.status === 'done').pop()
  1062→  1062→    const summary = `Готово за ${durS} с: ${run.nodes.length} узлов, ${dynamic} добавлено динамически, версий графа — ${run.version}. Итог: ${trunc(last?.output || '—', 220)}`
  1063→  1063→    run.summary = summary
  1064→  1064→    this.emitEvent(rt, { type: 'run:summary', ts: now(), summary })
  1065→  1065→    this.log(rt, 'run', `✓ Задача выполнена за ${durS} с`)
  1066→  1066→    store.save(run)
  1067→  1067→  }

  1069→  1069→  private recalcProgress(rt: Runtime) {
  1070→  1070→    const run = rt.snapshot
  1071→  1071→    // нереализованные ветки не считаются работой: прогресс по активным узлам
  1072→  1072→    const active = run.nodes.filter((n) => n.status !== 'discarded').length || 1
  1073→  1073→    const done = run.nodes.filter((n) => n.status === 'done').length
  1074→  1074→    run.progress = Math.round((done / active) * 100)
  1075→  1075→  }

  1077→  1077→  private downstreamOf(run: RunSnapshot, nodeId: string): GraphNode[] {
  1078→  1078→    const out: GraphNode[] = []
  1079→  1079→    const queue = [nodeId]
  1080→  1080→    const seen = new Set<string>()
  1081→  1081→    while (queue.length) {
  1082→  1082→      const cur = queue.shift()!
  1083→  1083→      for (const n of run.nodes) {
  1084→  1084→        if (n.dependsOn.includes(cur) && !seen.has(n.id)) {
  1085→  1085→          seen.add(n.id)
  1086→  1086→          out.push(n)
  1087→  1087→          queue.push(n.id)
  1088→  1088→        }
  1089→  1089→      }
  1090→  1090→    }
  1091→  1091→    return out
  1092→  1092→  }

  1094→  1094→  overrideNode(runId: string, nodeId: string, output: string): { ok: boolean; error?: string; resetCount?: number } {
  1095→  1095→    const rt = this.runs.get(runId)
  1096→  1096→    if (!rt) return { ok: false, error: 'Запуск не активен или уже выгружен' }
  1097→  1097→    const run = rt.snapshot
  1098→  1098→    const node = run.nodes.find((n) => n.id === nodeId && n.status === 'done')
  1099→  1099→    if (!node) return { ok: false, error: 'Узел не найден или ещё не завершён' }
  1100→  1100→    if (rt.stopped) return { ok: false, error: 'Запуск остановлен — переопределение недоступно' }
  1101→  1101→    const trimmed = (output || '').trim()
  1102→  1102→    if (!trimmed) return { ok: false, error: 'Результат не может быть пустым' }
  1103→  1103→
  1104→  1104→    node.output = trimmed.slice(0, 4000)
  1105→  1105→    node.overridden = true
  1106→  1106→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
  1107→  1107→    this.log(rt, 'intervention', `✎ Вмешательство: результат узла «${node.title}» переопределён пользователем`)
  1108→  1108→
  1109→  1109→    // нереализованные ветки не восстанавливаем — выбор на развилке остаётся в силе
  1110→  1110→    const down = this.downstreamOf(run, nodeId).filter((d) => d.status !== 'discarded')
  1111→  1111→    for (const d of down) {
  1112→  1112→      d.status = 'pending'
  1113→  1113→      d.output = undefined
  1114→  1114→      d.startedAt = undefined
  1115→  1115→      d.finishedAt = undefined
  1116→  1116→      d.durationMs = undefined
  1117→  1117→      d.error = undefined
  1118→  1118→      d.overridden = false
  1119→  1119→      this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...d } })
  1120→  1120→    }
  1121→  1121→    if (down.length) {
  1122→  1122→      this.log(rt, 'intervention', `↓ Пересчёт: ${down.length} зависимых узлов сброшены и будут выполнены заново`)
  1123→  1123→      this.recalcProgress(rt)
  1124→  1124→    }
  1125→  1125→    if (down.length && !rt.loopActive) {
  1126→  1126→      this.pushStatus(rt, 'running', run.progress)
  1127→  1127→      this.runLoop(rt).catch(() => {})
  1128→  1128→    }
  1129→  1129→    store.save(run)
  1130→  1130→    return { ok: true, resetCount: down.length }
  1131→  1131→  }

  1133→  1133→  retryNode(runId: string, nodeId: string): { ok: boolean; error?: string } {
  1134→  1134→    const rt = this.runs.get(runId)
  1135→  1135→    if (!rt) return { ok: false, error: 'Запуск не активен' }
  1136→  1136→    const node = rt.snapshot.nodes.find((n) => n.id === nodeId)
  1137→  1137→    if (!node || node.status !== 'error') return { ok: false, error: 'Узел не в состоянии ошибки' }
  1138→  1138→    rt.stopped = false
  1139→  1139→    node.status = 'pending'
  1140→  1140→    node.error = undefined
  1141→  1141→    node.startedAt = undefined
  1142→  1142→    node.finishedAt = undefined
  1143→  1143→    node.durationMs = undefined
  1144→  1144→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
  1145→  1145→    this.log(rt, 'intervention', `↻ Повтор узла «${node.title}» по запросу пользователя`)
  1146→  1146→    if (!rt.loopActive) {
  1147→  1147→      this.pushStatus(rt, 'running', rt.snapshot.progress)
  1148→  1148→      this.runLoop(rt).catch(() => {})
  1149→  1149→    }
  1150→  1150→    store.save(rt.snapshot)
  1151→  1151→    return { ok: true }
  1152→  1152→  }
