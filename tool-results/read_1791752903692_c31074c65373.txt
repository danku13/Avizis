   106→  async start(taskInput: string, mode: RunMode, decisionMode: DecisionMode = 'manual'): Promise<RunSnapshot> {
   107→    const task = (taskInput || '').trim() || DEFAULT_TASK
   108→    const id = 'run_' + now().toString(36) + Math.random().toString(36).slice(2, 7)
   109→    const scenario = mode === 'demo' ? detectDemoScenario(task) : 'data'
   110→    const run: RunSnapshot = {
   111→      id,
   112→      task,
   113→      mode,
   114→      decisionMode,
   115→      status: 'planning',
   116→      createdAt: now(),
   117→      progress: 0,
   118→      nodes: [],
   119→      version: 0,
   120→      versions: [],
   121→      events: [],
   122→      logs: [],
   123→    }
   124→    const rt: Runtime = {
   125→      snapshot: run,
   126→      stopped: false,
   127→      paused: false,
   128→      loopActive: false,
   129→      dynamicAdded: 0,
   130→      demoAdaptDone: new Set(),
   131→      scenario,
   132→      demoInsertionsDone: new Set(),
   133→      emergentAdded: 0,
   134→      decisionMode,
   135→      pendingDecision: null,
   136→    }
   137→    this.runs.set(id, rt)
   138→
   139→    this.emit('run:created', { runId: id, task, mode, decisionMode, createdAt: run.createdAt })
   140→    this.emitEvent(rt, { type: 'decision:mode', ts: now(), mode: decisionMode })
   141→    this.pushStatus(rt, 'planning', 0)
   142→    this.log(rt, 'run', `Запуск агента · ${mode === 'demo' ? 'демо-режим (симуляция)' : 'LLM через z-ai-web-dev-sdk'} · задача: «${trunc(task, 90)}»`)
   143→
   144→    if (mode === 'llm') {
   145→      this.log(rt, 'plan', 'Планировщик LLM анализирует задачу…')
   146→      try {
   147→        await getZai()
   148→      } catch (e: any) {
   149→        this.log(rt, 'run', `Ошибка инициализации SDK: ${e?.message || e}`)
   150→        this.pushStatus(rt, 'error', 0)
   151→        store.save(run)
   152→        return run
   153→      }
   154→    } else {
   155→      this.log(rt, 'plan', 'Планировщик строит граф задачи…')
   156→      await sleep(900)
   157→    }
   158→
   159→    let nodes: GraphNode[]
   160→    if (mode === 'llm') {
   161→      nodes = await this.planWithLLM(task, rt)
   162→    } else if (scenario === 'writing') {
   163→      nodes = demoWritingPlan()
   164→      this.log(rt, 'plan', 'Демо-сценарий «Напиши пост»: размытый запрос — агент будет уточнять вектор по ходу выполнения')
   165→    } else {
   166→      nodes = demoPlan()
   167→    }
   168→
   169→    this.emitEvent(rt, { type: 'graph:init', ts: now(), nodes })
   170→    run.nodes = nodes.map((n) => ({ ...n }))
   171→    run.version = 1
   172→    run.versions.push({ version: 1, createdAt: now(), description: 'Исходный план агента', nodeIds: nodes.map((n) => n.id) })
   173→    this.log(rt, 'plan', `План готов: ${nodes.length} узлов · версия графа v1`)
   174→
   175→    // стартовая самооценка «понимания задачи»: насколько узкий вектор решения
   176→    const initialCertainty = mode === 'llm' ? (rt.snapshot.certainty ?? 0.4) : DEMO_INITIAL_CERTAINTY[scenario]
   177→    const initialNote = mode === 'llm' ? undefined : DEMO_INITIAL_CERTAINTY_NOTE[scenario]
   178→    this.emitCertainty(rt, initialCertainty, initialNote)
   179→
   180→    this.pushStatus(rt, 'running', 0)
   181→    this.log(rt, 'run', 'Выполнение графа началось')
   182→    store.save(run)
   183→
   184→    this.runLoop(rt).catch((err: any) => {
   185→      this.log(rt, 'run', `Внутренняя ошибка цикла выполнения: ${err?.message || err}`)
   186→      this.pushStatus(rt, 'error', run.progress)
   187→    })
   188→    return run
   189→  }

   191→  private async planWithLLM(task: string, rt: Runtime): Promise<GraphNode[]> {
   192→    const user = `Задача пользователя: «${task}»
   193→
   194→Сначала ОЦЕНИ: если запрос размыт (не заданы формат, объём, тон, аудитория, скоуп или подход) — начни понимание с низкой оценки certainty и включи развилки-уточнения. Если задача про разработку — уточняй продуктовые и архитектурные вопросы; если про тексты — tone of voice и формат; если про данные/анализ — стратегию проверки.
   195→
   196→Составь план из 4–7 шагов с ОДНОЙ-ДВУМЯ развилками (узлами-решениями): точками, где выбор одного или нескольких вариантов меняет дальнейшие шаги и итоговый результат. Формат ответа — строго один JSON-объект без каких-либо дополнений:
   197→
   198→{"certainty":0.3,"nodes":[
   199→  {"id":"n1","title":"краткое название шага","description":"что делает шаг, 1-2 предложения","dependsOn":[]},
   200→  {"id":"dec1","title":"Решение: ...","description":"суть развилки","dependsOn":["n1"],
   201→   "decision":{"question":"вопрос, который решается на развилке","domain":"format","selection":"single","certaintyAfter":0.6,"options":[
   202→     {"id":"a","label":"название варианта","rationale":"обоснование: плюсы, минусы, цена выбора","steps":[
   203→       {"id":"ba1","title":"шаг ветки A","description":"..."}
   204→     ]},
   205→     {"id":"b","label":"название варианта","rationale":"...","steps":[
   206→       {"id":"bb1","title":"шаг ветки B","description":"..."}
   207→     ]}
   208→   ]}},
   209→  {"id":"nfin","title":"Формирование результата","description":"сведение итога","dependsOn":["ba1","bb1"]}
   210→]}
   211→
   212→Правила:
   213→- Названия и описания — на русском языке. Первый узел имеет пустой dependsOn.
   214→- certainty — число 0..1: твоя оценка понимания задачи ДО уточнений (размытый запрос → 0.2-0.4, конкретный → 0.6-0.8).
   215→- Развилок не больше двух; вариантов 2–4; шагов в каждой ветке 1–3. Первый шаг ветки выполняется вместо шагов других веток.
   216→- domain — один из: scope, specialization, product, architecture, tone, format, context, strategy.
   217→- selection — "single" (выбрать один вариант) или "mixed" (смешанное решение: допустимо выбрать несколько вариантов — например, блоки контента). certaintyAfter — оценка понимания ПОСЛЕ закрытия вопроса.

   220→    try {
   221→      const json = await chatJson(PLANNER_SYS, user, 60_000, 2)
   222→      const nodes = this.sanitizePlan(json, rt)
   223→      if (nodes) {
   224→        const decCount = nodes.filter((n) => n.kind === 'decision').length
   225→        this.log(rt, 'plan', `Планировщик LLM построил граф из ${nodes.length} узлов${decCount ? ` · развилок: ${decCount}` : ''}`)

   235→  /** Нормализация плана LLM: цепочки + узел-развилка с ветками вариантов + certainty */
   236→  private sanitizePlan(json: any, rt?: Runtime): GraphNode[] | null {
   237→    // стартовая самооценка «понимания задачи» от планировщика
   238→    if (rt) {
   239→      const c = Number(json?.certainty)
   240→      if (Number.isFinite(c)) rt.snapshot.certainty = Math.min(1, Math.max(0, c))
   241→    }
   242→    const rawNodes = Array.isArray(json?.nodes) ? json.nodes.slice(0, 12) : null
   243→    if (!rawNodes || rawNodes.length === 0) return null
   244→
   245→    const used = new Set<string>()
   246→    const cleanId = (v: any, fallback: string): string => {
   247→      let s = String(v ?? '').trim().toLowerCase().replace(/[^a-z0-9]/g, '').slice(0, 12)
   248→      if (!s) s = fallback
   249→      while (used.has(s)) s = (s.slice(0, 9) || 'n') + 'x' + Math.floor(Math.random() * 900 + 100)
   250→      used.add(s)
   251→      return s
   252→    }
   253→
   254→    const declared = new Set<string>()
   255→    const cleanDeps = (raw: any, exclude: Set<string>): string[] => {
   256→      const arr: any[] = Array.isArray(raw?.dependsOn) ? raw.dependsOn : []
   257→      const deps = arr
   258→        .map((d: any) => String(d ?? '').trim().toLowerCase().replace(/[^a-z0-9]/g, ''))
   259→        .filter((d: string) => d && declared.has(d) && !exclude.has(d))
   260→      return [...new Set(deps)].slice(0, 5)
   261→    }
   262→
   263→    const nodes: GraphNode[] = []
   264→    rawNodes.forEach((raw: any, i: number) => {
   265→      const id = cleanId(raw?.id, `n${i + 1}`)
   266→      const decisionRaw = raw?.decision
   267→      const optionsRaw = Array.isArray(decisionRaw?.options) ? decisionRaw.options.filter((o: any) => o).slice(0, 4) : []
   268→      const hasDecision = optionsRaw.length >= 2 && !!decisionRaw?.question
   269→
   270→      if (hasDecision) {
   271→        const branchIds = new Set<string>()
   272→        const selection: 'single' | 'mixed' = decisionRaw?.selection === 'mixed' ? 'mixed' : 'single'
   273→        const domain: DecisionDomain | undefined = DECISION_DOMAINS.includes(decisionRaw?.domain)
   274→          ? decisionRaw.domain
   275→          : undefined
   276→        const certaintyAfter = Number(decisionRaw?.certaintyAfter)
   277→        const options: DecisionOption[] = []
   278→        optionsRaw.forEach((ro: any, j: number) => {
   279→          const optId = ['a', 'b', 'c', 'd'][j]
   280→          const stepsRaw = Array.isArray(ro?.steps) ? ro.steps.filter((s: any) => s).slice(0, 3) : []
   281→          options.push({
   282→            id: optId,
   283→            label: String(ro?.label || `Вариант ${optId}`).slice(0, 60),
   284→            rationale: String(ro?.rationale || ''),
   285→            nextSteps: stepsRaw.map((s: any) => String(s?.title || 'шаг ветки').slice(0, 60)),
   286→          })
   287→          // шаги ветки: цепочка от развилки
   288→          let prev = id
   289→          stepsRaw.forEach((rs: any, k: number) => {
   290→            const bid = cleanId(rs?.id, `${id}${optId}${k + 1}`)
   291→            branchIds.add(bid)
   292→            declared.add(bid)
   293→            nodes.push({
   294→              id: bid,
   295→              title: String(rs?.title || `Шаг ветки ${optId}${k + 1}`).slice(0, 60),
   296→              description: String(rs?.description || ''),
   297→              status: 'pending',
   298→              dependsOn: [prev],
   299→              dynamic: false,
   300→              runs: 0,
   301→              branchOf: { decisionId: id, optionId: optId },
   302→            })
   303→            prev = bid
   304→          })
   305→        })
   306→        declared.add(id)
   307→        nodes.push({
   308→          id,
   309→          kind: 'decision',
   310→          title: String(raw?.title || 'Решение').slice(0, 60),
   311→          description: String(raw?.description || ''),
   312→          status: 'pending',
   313→          dependsOn: cleanDeps(raw, branchIds),
   314→          dynamic: false,
   315→          runs: 0,
   316→          decision: {
   317→            question: String(decisionRaw.question),
   318→            domain,
   319→            selection,
   320→            certaintyAfter: Number.isFinite(certaintyAfter) ? Math.min(1, Math.max(0, certaintyAfter)) : undefined,
   321→            options,
   322→            status: 'pending',
   323→          },
   324→        })
   325→      } else {
   326→        declared.add(id)
   327→        nodes.push({
   328→          id,
   329→          title: String(raw?.title || `Шаг ${i + 1}`).slice(0, 60),
   330→          description: String(raw?.description || ''),
   331→          status: 'pending',
   332→          dependsOn: cleanDeps(raw, new Set()),
   333→          dynamic: false,
   334→          runs: 0,
   335→        })
   336→      }
   337→    })
   338→    if (nodes.length < 2) return null
   339→    return nodes
   340→  }

   344→  private async runLoop(rt: Runtime) {
   345→    if (rt.loopActive) return
   346→    rt.loopActive = true
   347→    try {
   348→      while (true) {
   349→        const run = rt.snapshot
   350→        if (rt.stopped) {
   351→          this.finalizeStopped(rt)
   352→          return
   353→        }
   354→        const node = this.nextReadyNode(run)
   355→        if (!node) {
   356→          const errored = run.nodes.find((n) => n.status === 'error')
   357→          if (errored) {
   358→            this.log(rt, 'node-error', `Узел «${errored.title}» завершился ошибкой — выполнение остановлено. Доступно вмешательство: «Повторить узел».`)
   359→            this.pushStatus(rt, 'error', run.progress)
   360→            run.finishedAt = now()
   361→            store.save(run)
   362→            return
   363→          }
   364→          const pending = run.nodes.filter((n) => n.status === 'pending')
   365→          if (pending.length) {
   366→            this.log(rt, 'run', 'Обнаружена циклическая зависимость — оставшиеся узлы пропущены')
   367→            pending.forEach((n) => {
   368→              n.status = 'skipped'
   369→              this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...n } })
   370→            })
   371→          }
   372→          this.finalizeCompleted(rt)
   373→          return
   374→        }
   375→
   376→        await this.waitWhilePaused(rt)
   377→        if (rt.stopped) continue
   378→
   379→        node.input = this.buildInput(run, node)
   380→        node.status = 'running'
   381→        node.startedAt = now()
   382→        node.runs = (node.runs ?? 0) + 1
   383→        node.error = undefined
   384→        this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   385→        this.log(rt, 'node-start', `▶ «${node.title}» — выполнение`)
   386→
   387→        // ─── Узел-развилка: рассуждение → предложение вариантов → выбор → гашение веток ───
   388→        if (node.kind === 'decision' && node.decision) {
   389→          const decided = await this.handleDecisionNode(rt, node)
   390→          if (!decided) continue // остановлен во время решения — цикл завершит очистку
   391→          this.recalcProgress(rt)
   392→          this.pushStatus(rt, rt.paused ? 'paused' : 'running', run.progress)
   393→          store.save(run)
   394→          continue
   395→        }
   396→
   397→        try {
   398→          const output =
   399→            run.mode === 'demo' ? await this.demoExecute(node, rt) : await this.llmExecute(run, node)
   400→          node.status = 'done'
   401→          node.output = output
   402→          node.finishedAt = now()
   403→          node.durationMs = node.finishedAt - (node.startedAt ?? node.finishedAt)
   404→          this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   405→          this.log(rt, 'node-done', `✓ «${node.title}» — завершён за ${((node.durationMs ?? 0) / 1000).toFixed(1)} с`)
   406→        } catch (err: any) {
   407→          if (err instanceof StoppedError) {
   408→            node.status = 'pending'
   409→            node.startedAt = undefined
   410→            node.input = undefined
   411→            this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   412→            this.finalizeStopped(rt)
   413→            return
   414→          }
   415→          node.status = 'error'
   416→          node.error = err?.message || String(err)
   417→          node.finishedAt = now()
   418→          node.durationMs = node.finishedAt - (node.startedAt ?? node.finishedAt)
   419→          this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   420→          this.log(rt, 'node-error', `✗ «${node.title}» — ошибка: ${trunc(node.error ?? '', 140)}`)
   421→          continue
   422→        }
   423→
   424→        await this.maybeAdapt(rt, node)
   425→        this.recalcProgress(rt)
   426→        this.pushStatus(rt, rt.paused ? 'paused' : 'running', run.progress)
   427→        store.save(run)
   428→      }
   429→    } finally {
   430→      rt.loopActive = false
   431→    }
   432→  }

   434→  private nextReadyNode(run: RunSnapshot): GraphNode | undefined {
   435→    const byId = new Map(run.nodes.map((n) => [n.id, n]))
   436→    // «discarded» = ветка не реализована: зависимость от неё считается удовлетворённой
   437→    const ready = (n: GraphNode) =>
   438→      n.dependsOn.every((d) => {
   439→        const dep = byId.get(d)
   440→        return !dep || dep.status === 'done' || dep.status === 'skipped' || dep.status === 'discarded'
   441→      })
   442→    return run.nodes.find((n) => n.status === 'pending' && ready(n))
   443→  }

   498→  private async maybeAdapt(rt: Runtime, finished: GraphNode) {
   499→    const run = rt.snapshot
   500→    if (run.mode === 'demo') {
   501→      if (rt.scenario === 'data') {
   502→        const idx = DEMO_ADAPTATIONS.findIndex((a, i) => a.after === finished.id && !rt.demoAdaptDone.has(i))
   503→        if (idx < 0) return
   504→        rt.demoAdaptDone.add(idx)
   505→        await sleep(450)
   506→        const spec = DEMO_ADAPTATIONS[idx]
   507→        this.insertNode(rt, { ...spec.node, dependsOn: [finished.id] }, spec.reason)
   508→        return
   509→      }
   510→      // сценарий «Напиши пост»: агент анализирует результаты шагов и вставляет развилки-уточнения
   511→      const insIdx = DEMO_DECISION_INSERTIONS.findIndex(
   512→        (d, i) => d.afterAny.includes(finished.id) && !rt.demoInsertionsDone.has(i)
   513→      )
   514→      if (insIdx < 0) return
   515→      rt.demoInsertionsDone.add(insIdx)
   516→      await sleep(450)
   517→      const spec = DEMO_DECISION_INSERTIONS[insIdx]
   518→      this.insertDecision(rt, finished.id, spec.decision, spec.reason)
   519→      return
   520→    }
   521→
   522→    // LLM-режим: супервайзер анализирует результаты шагов — новый шаг / вопрос-развилка / ничего.
   523→    // Количество эмерджентных вопросов и динамических вставок НЕ ограничено (ADR-011):
   524→    // дисциплину задаёт промпт (полезность вопроса, недублирование), а не жёсткие счётчики.
   525→    const hasPendingSuccessors = run.nodes.some((n) => n.status === 'pending' && n.dependsOn.includes(finished.id))
   526→    const anyPending = run.nodes.some((n) => n.status === 'pending')
   527→    if (!hasPendingSuccessors && !anyPending) return
   528→    // вопрос не вставляется, пока открыта другая развилка: уточнения идут последовательно
   529→    const hasOpenDecision = run.nodes.some((n) => n.kind === 'decision' && (n.status === 'pending' || n.status === 'running'))
   530→    const askAllowed = !hasOpenDecision
   531→
   532→    const graphDesc = run.nodes
   533→      .map((n) => `- ${n.id} «${n.title}» [${n.status}]${n.output ? ': ' + trunc(n.output, 400) : ''}`)
   534→      .join('\n')
   535→    const openQuestions = run.nodes
   536→      .filter((n) => n.kind === 'decision')
   537→      .map((n) => n.decision ? `«${n.title}» (${n.decision.status})` : '')
   538→      .filter(Boolean)
   539→      .join(', ')
   540→    const user = `Задача: «${run.task}»
   541→
   542→Текущий граф:
   543→${graphDesc}
   544→
   545→Заданные ранее вопросы-развилки: ${openQuestions || '—'}
   546→
   547→Только что завершён узел «${finished.title}»: ${trunc(finished.output || '', 2000)}
   548→
   549→Проанализируй входящие данные завершённого шага и реши:
   550→1) add_node — если есть ЯВНЫЙ пробел, закрываемый обычным шагом (например, данные требуют валидации);
   551→2) ask — если обнаружена неоднозначность, влияющая на итог (расширение/сужение скоупа, специализация, продуктовые или архитектурные вопросы для разработки, tone of voice или формат для текстов, контекст задачи): предложи вопрос с вариантами и обоснованием;${
   552→      askAllowed ? '' : ' (сейчас ask недоступен — используй только no_changes/add_node)'
   553→    }
   554→3) no_changes — в остальных случаях (в подавляющем большинстве).
   555→Количество вопросов не ограничено, но каждый обязан быть полезным: не дублируй уже заданные/закрытые вопросы (список выше), не уточняй то, что не влияет на итог. Во всех текстах (description, rationale, reason) не раздувай контекст: только полезная информация, без воды и повторов.
   556→Также верни certainty — текущую оценку понимания задачи (0..1: насколько узкий вектор решения сложился).
   557→
   558→Формат ответа:
   559→{"decision":"no_changes","certainty":0.6}
   560→или
   561→{"decision":"add_node","certainty":0.6,"node":{"title":"...","description":"...","reason":"почему добавляем"}}
   562→или
   563→{"decision":"ask","certainty":0.5,"question":{"question":"текст вопроса","domain":"scope","selection":"single","certaintyAfter":0.8,"reason":"почему вопрос важен","options":[{"id":"a","label":"название","rationale":"обоснование","steps":[{"title":"шаг ветки","description":"..."}]}]}}
   564→
   565→Домены: scope, specialization, product, architecture, tone, format, context, strategy. selection: single (один вариант) или mixed (смешанное решение из нескольких). Вариантов 2–4, шагов в ветке 1–3.`
   566→
   567→    try {
   568→      const json = await chatJson(SUPERVISOR_SYS, user, 45_000, 0)
   569→      if (typeof json?.certainty === 'number' && Number.isFinite(json.certainty)) {
   570→        const c = Math.min(1, Math.max(0, json.certainty))
   571→        if (Math.abs(c - (run.certainty ?? 0)) > 0.05) this.emitCertainty(rt, c)
   572→      }
   573→      if (json?.decision === 'add_node' && json.node?.title) {
   574→        const id = `dyn${rt.dynamicAdded + 1}`
   575→        this.insertNode(
   576→          rt,
   577→          {
   578→            id,
   579→            title: String(json.node.title).slice(0, 60),
   580→            description: String(json.node.description || ''),
   581→            dependsOn: [finished.id],
   582→          },
   583→          String(json.node.reason || 'Выявлен пробел в плане')
   584→        )
   585→      } else if (json?.decision === 'ask' && askAllowed) {
   586→        const q = json.question
   587→        if (q?.question && Array.isArray(q?.options) && q.options.length >= 2) {
   588→          const options = q.options
   589→            .filter((o: any) => o?.label)
   590→            .slice(0, 4)
   591→            .map((o: any, j: number) => ({
   592→              id: ['a', 'b', 'c', 'd'][j],
   593→              label: String(o.label).slice(0, 60),
   594→              rationale: String(o.rationale || ''),
   595→              steps: (Array.isArray(o.steps) ? o.steps.filter((s: any) => s).slice(0, 3) : []).map((s: any) => ({
   596→                title: String(s?.title || 'шаг ветки').slice(0, 60),
   597→                description: String(s?.description || ''),
   598→              })),
   599→            }))
   600→          if (options.length >= 2) {
   601→            const certaintyAfter = Number(q.certaintyAfter)
   602→            this.insertDecision(
   603→              rt,
   604→              finished.id,
   605→              {
   606→                id: `decq${rt.emergentAdded + 1}`,
   607→                title: `Вопрос: ${String(q.question).slice(0, 40)}`,
   608→                description: String(q.reason || 'Выявлена неоднозначность, влияющая на итог'),
   609→                question: String(q.question),
   610→                domain: DECISION_DOMAINS.includes(q.domain) ? q.domain : 'context',
   611→                selection: q.selection === 'mixed' ? 'mixed' : 'single',
   612→                certaintyAfter: Number.isFinite(certaintyAfter)
   613→                  ? Math.min(1, Math.max(0, certaintyAfter))
   614→                  : Math.min(0.95, (run.certainty ?? 0.4) + 0.2),
   615→                options,
   616→              },
   617→              String(q.reason || 'Агент обнаружил неоднозначность, влияющую на итог')
   618→            )
   619→          }
   620→        }
   621→      }
   622→    } catch {
   623→      /* супервайзер недоступен → план не меняем */
   624→    }
   625→  }

   627→  private insertNode(rt: Runtime, spec: { id: string; title: string; description: string; dependsOn: string[] }, reason: string) {
   628→    const run = rt.snapshot
   629→    const newNode: GraphNode = {
   630→      id: spec.id,
   631→      title: spec.title,
   632→      description: spec.description,
   633→      status: 'pending',
   634→      dependsOn: [...spec.dependsOn],
   635→      dynamic: true,
   636→      reason,
   637→      addedAtVersion: run.version + 1,
   638→      runs: 0,
   639→    }
   640→    const changed: GraphNode[] = [newNode]
   641→    const anchor = spec.dependsOn[0]
   642→    // Перепривязываем ещё не начатые узлы, зависящие от якоря, на новый узел
   643→    for (const n of run.nodes) {
   644→      if (n.status === 'pending' && n.dependsOn.includes(anchor)) {
   645→        n.dependsOn = [...new Set(n.dependsOn.map((d) => (d === anchor ? newNode.id : d)))]
   646→        changed.push({ ...n })
   647→      }
   648→    }
   649→    run.nodes.push(newNode)
   650→    run.version += 1
   651→    rt.dynamicAdded += 1
   652→    run.versions.push({ version: run.version, createdAt: now(), description: reason, nodeIds: run.nodes.map((n) => n.id) })
   653→    this.emitEvent(rt, { type: 'graph:adapt', ts: now(), reason, version: run.version, nodes: changed.map((n) => ({ ...n })) })
   654→    this.log(rt, 'adapt', `＋ Граф адаптирован (v${run.version}): добавлен узел «${newNode.title}». Причина: ${reason}`)
   655→    store.save(run)
   656→  }

   658→  /**
   659→   * Вставка эмерджентной развилки: агент проанализировал результаты шага и
   660→   * обнаружил неоднозначность, влияющую на итог. Создаёт узел-вопрос с ветками
   661→   * вариантов; ещё не начатые узлы, ждавшие якорь, переводятся на хвосты веток —
   662→   * итог корректно дождётся только реализованных веток.
   663→   */
   664→  private insertDecision(
   665→    rt: Runtime,
   666→    anchorId: string,
   667→    spec: {
   668→      id: string
   669→      title: string
   670→      description: string
   671→      question: string
   672→      domain?: string
   673→      selection: 'single' | 'mixed'
   674→      certaintyAfter?: number
   675→      options: Array<{ id: string; label: string; rationale: string; steps: Array<{ id?: string; title: string; description: string }> }>
   676→    },
   677→    reason: string
   678→  ) {
   679→    const run = rt.snapshot
   680→    const domain: DecisionDomain | undefined = DECISION_DOMAINS.includes(spec.domain as DecisionDomain)
   681→      ? (spec.domain as DecisionDomain)
   682→      : undefined
   683→    const version = run.version + 1
   684→
   685→    const decNode: GraphNode = {
   686→      id: spec.id,
   687→      kind: 'decision',
   688→      title: spec.title,
   689→      description: spec.description,
   690→      status: 'pending',
   691→      dependsOn: [anchorId],
   692→      dynamic: true,
   693→      reason,
   694→      addedAtVersion: version,
   695→      runs: 0,
   696→      decision: {
   697→        question: spec.question,
   698→        domain,
   699→        selection: spec.selection,
   700→        certaintyAfter: spec.certaintyAfter,
   701→        options: spec.options.map((o) => ({
   702→          id: o.id,
   703→          label: o.label,
   704→          rationale: o.rationale,
   705→          nextSteps: o.steps.map((s) => s.title),
   706→        })),
   707→        status: 'pending',
   708→      },
   709→    }
   710→    const changed: GraphNode[] = [decNode]
   711→    const tails: string[] = []
   712→
   713→    // ветки вариантов: цепочки от развилки
   714→    spec.options.forEach((o) => {
   715→      let prev = spec.id
   716→      o.steps.forEach((s, k) => {
   717→        const bid = s.id || `${spec.id}${o.id}${k + 1}`
   718→        const branchNode: GraphNode = {
   719→          id: bid,
   720→          title: s.title,
   721→          description: s.description,
   722→          status: 'pending',
   723→          dependsOn: [prev],
   724→          dynamic: true,
   725→          reason,
   726→          addedAtVersion: version,
   727→          runs: 0,
   728→          branchOf: { decisionId: spec.id, optionId: o.id },
   729→        }
   730→        run.nodes.push(branchNode)
   731→        changed.push({ ...branchNode })
   732→        prev = bid
   733→      })
   734→      if (prev === spec.id) return // у варианта нет шагов — ветка «прямое продолжение»
   735→      tails.push(prev)
   736→    })
   737→    run.nodes.push(decNode)
   738→
   739→    // перепривязка: ещё не начатые узлы, ждавшие якорь, ждут хвосты ВСЕХ веток
   740→    // (невыбранные ветки станут discarded — зависимость удовлетворится автоматически)
   741→    for (const n of run.nodes) {
   742→      if (n.id !== decNode.id && n.status === 'pending' && n.dependsOn.includes(anchorId)) {
   743→        n.dependsOn = [...new Set(n.dependsOn.flatMap((d) => (d === anchorId ? (tails.length ? tails : [anchorId]) : [d])))].slice(0, 12)
   744→        changed.push({ ...n })
   745→      }
   746→    }
   747→
   748→    run.version = version
   749→    rt.dynamicAdded += 1
   750→    rt.emergentAdded += 1
   751→    run.versions.push({ version, createdAt: now(), description: reason, nodeIds: run.nodes.map((n) => n.id) })
   752→    this.emitEvent(rt, { type: 'graph:adapt', ts: now(), reason, version, nodes: changed.map((n) => ({ ...n })) })
   753→    this.log(
   754→      rt,
   755→      'adapt',
   756→      `＋ Эмерджентная развилка (v${version}): «${spec.title}» — ${spec.options.length} варианта${
   757→        spec.selection === 'mixed' ? ' (смешанный выбор допустим)' : ''
   758→      }. Причина: ${reason}`
   759→    )
   760→    store.save(run)
   761→  }

   763→  /** Публикация самооценки «понимания задачи»: насколько узкий вектор решения */
   764→  private emitCertainty(rt: Runtime, certainty: number, note?: string) {
   765→    const c = Math.min(1, Math.max(0, Number.isFinite(certainty) ? certainty : 0))
   766→    rt.snapshot.certainty = c
   767→    this.emitEvent(rt, { type: 'run:certainty', ts: now(), certainty: c, note })
   768→    const pct = Math.round(c * 100)
   769→    this.log(rt, 'decision', `◎ Понимание задачи: ${pct}% — ${note || 'агент переоценил вектор решения'}`)
   770→  }

   774→  /** Жизненный цикл узла-развилки: рассуждение → предложение вариантов → выбор → активация ветки */
   775→  private async handleDecisionNode(rt: Runtime, node: GraphNode): Promise<boolean> {
   776→    const dec = node.decision!
   777→    this.log(rt, 'decision', `⎇ Развилка «${node.title}»: ${dec.question}`)
   778→    try {
   779→      await this.decisionThinkDelay(rt)
   780→    } catch {
   781→      return this.rollbackDecision(rt, node)
   782→    }
   783→    if (rt.stopped) return this.rollbackDecision(rt, node)
   784→
   785→    dec.status = 'offered'
   786→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   787→    this.emitEvent(rt, {
   788→      type: 'decision:offer',
   789→      ts: now(),
   790→      nodeId: node.id,
   791→      question: dec.question,
   792→      options: dec.options,
   793→    })
   794→
   795→    if (rt.decisionMode === 'manual') {
   796→      this.log(
   797→        rt,
   798→        'decision',
   799→        `⎇ Режим «ручной выбор»: агент предлагает ${dec.options.length} варианта с обоснованием и ожидает решения пользователя${
   800→          dec.selection === 'mixed' ? ' (допустимо смешанное решение из нескольких)' : ''
   801→        }`
   802→      )
   803→      const choice = await this.waitForDecision(rt, node)
   804→      if (!choice) return this.rollbackDecision(rt, node)
   805→      this.resolveDecision(rt, node, choice.optionIds, choice.by, choice.reasoning)
   806→    } else {
   807→      this.log(rt, 'decision', '⎇ Режим «авто»: агент взвешивает варианты и принимает решение самостоятельно')
   808→      const picked = await this.autoDecide(rt, node)
   809→      if (rt.stopped) return this.rollbackDecision(rt, node)
   810→      this.resolveDecision(rt, node, picked.optionIds, 'agent', picked.reasoning)
   811→    }
   812→    return true
   813→  }

   815→  /** Откат развилки (остановка во время решения): узел возвращается в очередь */
   816→  private rollbackDecision(rt: Runtime, node: GraphNode): boolean {
   817→    node.status = 'pending'
   818→    node.startedAt = undefined

   843→  /** Авто-режим: демо — скриптованный выбор; LLM — вызов модуля принятия решений */
   844→  private async autoDecide(rt: Runtime, node: GraphNode): Promise<{ optionIds: string[]; reasoning: string }> {
   845→    const dec = node.decision!
   846→    if (rt.snapshot.mode === 'demo') {
   847→      await sleep(1200)
   848→      const scripted = DEMO_WRITING_AUTO_PICKS[node.id]
   849→      if (scripted) return { optionIds: [...scripted.optionIds], reasoning: scripted.reasoning }
   850→      return { optionIds: [...DEMO_AUTO_PICK.optionIds], reasoning: DEMO_AUTO_PICK.reasoning }
   851→    }
   852→    const mixed = dec.selection === 'mixed'
   853→    const user = `Задача: «${rt.snapshot.task}»
   854→
   855→Развилка: ${dec.question}
   856→Тип выбора: ${mixed ? 'mixed — можно выбрать НЕСКОЛЬКО вариантов (смешанное решение)' : 'single — нужно выбрать ровно один вариант'}
   857→
   858→Варианты:
   859→${dec.options.map((o) => `- id «${o.id}» — ${o.label}: ${o.rationale}`).join('\n')}
   860→
   861→${node.input || ''}
   862→
   863→${mixed ? 'Выбери один или несколько вариантов (id из списка) и обоснуй состав.' : 'Выбери один вариант (id из списка) и обоснуй выбор.'} Ответ строго JSON:
   864→${mixed ? '{"optionIds":["<id1>","<id2>"],"reasoning":"краткое, но исчерпывающее обоснование без воды"}' : '{"optionIds":["<id>"],"reasoning":"краткое, но исчерпывающее обоснование без воды"}'}`
   865→    try {
   866→      const json = await chatJson(DECIDER_SYS, user, 45_000, 1)
   867→      const rawIds: any[] = Array.isArray(json?.optionIds)
   868→        ? json.optionIds
   869→        : Array.isArray(json?.options)
   870→          ? json.options
   871→          : json?.optionId
   872→            ? [json.optionId]
   873→            : []
   874→      const valid = rawIds
   875→        .map((v) => String(v ?? '').trim().toLowerCase())
   876→        .filter((v) => dec.options.some((o) => o.id === v))
   877→      const unique = [...new Set(valid)]
   878→      if (unique.length && json?.reasoning) {
   879→        const picked = mixed ? unique : unique.slice(0, 1)
   880→        return { optionIds: picked, reasoning: String(json.reasoning) }
   881→      }
   882→    } catch {
   883→      /* fallback ниже */
   884→    }
   885→    return { optionIds: [dec.options[0].id], reasoning: 'Обоснование LLM недоступно — выбран первый вариант по умолчанию' }
   886→  }

   888→  /** Применяет выбор (один или несколько вариантов): активирует ветки, гасит альтернативы, фиксирует аудит-трейл */
   889→  private resolveDecision(rt: Runtime, node: GraphNode, optionIds: string[], by: 'agent' | 'user', reasoning?: string) {
   890→    const run = rt.snapshot
   891→    const dec = node.decision!
   892→    const single = dec.selection !== 'mixed'
   893→    // валидация: только известные id; single — ровно один
   894→    let picked = optionIds.map((id) => id.trim().toLowerCase()).filter((id) => dec.options.some((o) => o.id === id))
   895→    picked = [...new Set(picked)]
   896→    if (!picked.length) picked = [dec.options[0].id]
   897→    if (single) picked = picked.slice(0, 1)
   898→    const opts = picked.map((id) => dec.options.find((o) => o.id === id)!)
   899→    const isMixedResult = picked.length > 1
   900→    const pickedLabel = opts.map((o) => `«${o.label}»`).join(isMixedResult ? ' + ' : '')
   901→
   902→    node.status = 'done'
   903→    node.finishedAt = now()
   904→    node.durationMs = node.finishedAt - (node.startedAt ?? node.finishedAt)
   905→    node.decision = { ...dec, status: 'resolved', chosen: picked, chosenBy: by, reasoning: reasoning?.trim() || undefined }
   906→    node.output =
   907→      by === 'agent'
   908→        ? `${isMixedResult ? `Смешанное решение агента: ${pickedLabel}.` : `Решение агента: вариант ${pickedLabel}.`} ${reasoning || ''}`.trim()
   909→        : `${isMixedResult ? `Смешанный выбор пользователя: ${pickedLabel}.` : `Выбор пользователя: вариант ${pickedLabel}.`} ${opts.map((o) => o.rationale).join(' ')}`
   910→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   911→
   912→    // мгновенная визуализация последствий: нереализованные ветки гаснут
   913→    let discarded = 0
   914→    for (const n of run.nodes) {
   915→      if (n.branchOf?.decisionId === node.id && !picked.includes(n.branchOf.optionId) && n.status === 'pending') {
   916→        n.status = 'discarded'
   917→        discarded++
   918→        this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...n } })
   919→      }
   920→    }
   921→
   922→    this.emitEvent(rt, {
   923→      type: 'decision:resolved',
   924→      ts: now(),
   925→      nodeId: node.id,
   926→      optionIds: picked,
   927→      by,
   928→      reasoning: node.decision.reasoning,
   929→    })
   930→    this.log(
   931→      rt,
   932→      'decision',
   933→      `⎇ Решение принято (${by === 'agent' ? 'агентом' : 'пользователем'}): ${pickedLabel}${
   934→        isMixedResult ? ' — ветки активированы' : ' — ветка активирована'
   935→      }, узлов в нереализованных ветках: ${discarded}`
   936→    )
   937→
   938→    // закрытие вопроса сужает вектор понимания задачи
   939→    const after =
   940→      dec.certaintyAfter ?? Math.min(0.95, (rt.snapshot.certainty ?? 0.4) + 0.15)
   941→    this.emitCertainty(
   942→      rt,
   943→      Math.max(rt.snapshot.certainty ?? 0, after),
   944→      isMixedResult
   945→        ? `смешанное решение по вопросу «${trunc(node.title, 48)}»`
   946→        : `вопрос «${trunc(node.title, 48)}» закрыт`
   947→    )
   948→  }

   950→  /** Переключение режима решений на ходу; «авто» во время ожидания — агент решает немедленно */
   951→  setDecisionMode(runId: string, mode: DecisionMode): { ok: boolean; error?: string } {
   952→    const rt = this.runs.get(runId)
   953→    if (!rt) return { ok: false, error: 'Запуск не активен или уже выгружен' }
   954→    if (rt.decisionMode === mode) return { ok: true }
   955→    rt.decisionMode = mode
   956→    rt.snapshot.decisionMode = mode
   957→    this.emitEvent(rt, { type: 'decision:mode', ts: now(), mode })
   958→    this.log(
   959→      rt,
   960→      'control',
   961→      `⎇ Режим решений: ${mode === 'auto' ? '«авто» — агент решает сам' : '«ручной выбор» — агент предлагает варианты и ждёт'}`
   962→    )
   963→    store.save(rt.snapshot)
   964→    // если сейчас ожидается выбор, а включили «авто» — агент принимает решение вместо пользователя
   965→    if (mode === 'auto' && rt.pendingDecision) {
   966→      const pd = rt.pendingDecision
   967→      rt.pendingDecision = null
   968→      const node = rt.snapshot.nodes.find((n) => n.id === pd.nodeId)
   969→      if (node?.decision) {
   970→        this.log(rt, 'decision', '⎇ Ожидание выбора прервано: агент принимает решение вместо пользователя')
   971→        this.autoDecide(rt, node)
   972→          .then((picked) => pd.resolve({ optionIds: picked.optionIds, by: 'agent', reasoning: picked.reasoning }))
   973→          .catch(() => pd.resolve(null))
   974→      } else {
   975→        pd.resolve(null)
   976→      }
   977→    }
   978→    return { ok: true }
   979→  }

   981→  /** Выбор пользователя на развилке (из модалки): один или несколько вариантов */
   982→  resolveDecisionFromUser(runId: string, nodeId: string, optionIds: string[]): { ok: boolean; error?: string } {
   983→    const rt = this.runs.get(runId)
   984→    if (!rt) return { ok: false, error: 'Запуск не активен' }
   985→    const pd = rt.pendingDecision
   986→    if (!pd || pd.nodeId !== nodeId) return { ok: false, error: 'Решение не ожидается (уже принято или запуск завершён)' }
   987→    const node = rt.snapshot.nodes.find((n) => n.id === nodeId)
   988→    const dec = node?.decision
   989→    const ids = (optionIds || []).map((id) => String(id).trim().toLowerCase())
   990→    if (!dec || !ids.length || !ids.every((id) => dec.options.some((o) => o.id === id))) {
   991→      return { ok: false, error: 'Неизвестный вариант' }
   992→    }
   993→    rt.pendingDecision = null
   994→    pd.resolve({ optionIds: ids, by: 'user' })
   995→    return { ok: true }
   996→  }

  1000→  control(runId: string, action: 'pause' | 'resume' | 'stop') {
  1001→    const rt = this.runs.get(runId)
  1002→    if (!rt) return
  1003→    if (action === 'pause') {
  1004→      if (!rt.paused && !rt.stopped && rt.snapshot.status === 'running') {
  1005→        rt.paused = true
  1006→        this.pushStatus(rt, 'paused', rt.snapshot.progress)
  1007→        this.log(rt, 'control', '⏸ Пауза: текущий узел завершится, новые не стартуют')
  1008→      }
  1009→    } else if (action === 'resume') {
  1010→      if (rt.paused) {
  1011→        rt.paused = false
  1012→        this.log(rt, 'control', '▶ Выполнение возобновлено')
  1013→        if (rt.snapshot.status === 'paused') {
  1014→          this.pushStatus(rt, 'running', rt.snapshot.progress)
  1015→        }
  1016→      }
  1017→    } else if (action === 'stop') {
  1018→      if (rt.stopped) return
  1019→      if (['completed', 'error', 'stopped'].includes(rt.snapshot.status)) return
  1020→      rt.stopped = true
  1021→      rt.paused = false
  1022→      // если ждём выбор пользователя на развилке — разблокируем ожидание
  1023→      if (rt.pendingDecision) {
  1024→        const pd = rt.pendingDecision
  1025→        rt.pendingDecision = null
  1026→        pd.resolve(null)
  1027→      }
  1028→      this.log(rt, 'control', '⏹ Остановка запрошена — применится после завершения текущего узла')
  1029→    }
  1030→  }

  1070→  private recalcProgress(rt: Runtime) {
  1071→    const run = rt.snapshot
  1072→    // нереализованные ветки не считаются работой: прогресс по активным узлам
  1073→    const active = run.nodes.filter((n) => n.status !== 'discarded').length || 1
  1074→    const done = run.nodes.filter((n) => n.status === 'done').length
  1075→    run.progress = Math.round((done / active) * 100)
  1076→  }

  1078→  private downstreamOf(run: RunSnapshot, nodeId: string): GraphNode[] {
  1079→    const out: GraphNode[] = []
  1080→    const queue = [nodeId]
  1081→    const seen = new Set<string>()
  1082→    while (queue.length) {
  1083→      const cur = queue.shift()!
  1084→      for (const n of run.nodes) {
  1085→        if (n.dependsOn.includes(cur) && !seen.has(n.id)) {
  1086→          seen.add(n.id)
  1087→          out.push(n)
  1088→          queue.push(n.id)
  1089→        }
  1090→      }
  1091→    }
  1092→    return out
  1093→  }

  1095→  overrideNode(runId: string, nodeId: string, output: string): { ok: boolean; error?: string; resetCount?: number } {
  1096→    const rt = this.runs.get(runId)
  1097→    if (!rt) return { ok: false, error: 'Запуск не активен или уже выгружен' }
  1098→    const run = rt.snapshot
  1099→    const node = run.nodes.find((n) => n.id === nodeId && n.status === 'done')
  1100→    if (!node) return { ok: false, error: 'Узел не найден или ещё не завершён' }
  1101→    if (rt.stopped) return { ok: false, error: 'Запуск остановлен — переопределение недоступно' }
  1102→    const trimmed = (output || '').trim()
  1103→    if (!trimmed) return { ok: false, error: 'Результат не может быть пустым' }
  1104→
  1105→    node.output = trimmed
  1106→    node.overridden = true
  1107→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
  1108→    this.log(rt, 'intervention', `✎ Вмешательство: результат узла «${node.title}» переопределён пользователем`)
  1109→
  1110→    // нереализованные ветки не восстанавливаем — выбор на развилке остаётся в силе
  1111→    const down = this.downstreamOf(run, nodeId).filter((d) => d.status !== 'discarded')
  1112→    for (const d of down) {
  1113→      d.status = 'pending'
  1114→      d.output = undefined
  1115→      d.startedAt = undefined
  1116→      d.finishedAt = undefined
  1117→      d.durationMs = undefined
  1118→      d.error = undefined
  1119→      d.overridden = false
  1120→      this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...d } })
  1121→    }
  1122→    if (down.length) {
  1123→      this.log(rt, 'intervention', `↓ Пересчёт: ${down.length} зависимых узлов сброшены и будут выполнены заново`)
  1124→      this.recalcProgress(rt)
  1125→    }
  1126→    if (down.length && !rt.loopActive) {
  1127→      this.pushStatus(rt, 'running', run.progress)
  1128→      this.runLoop(rt).catch(() => {})
  1129→    }
  1130→    store.save(run)
  1131→    return { ok: true, resetCount: down.length }
  1132→  }
