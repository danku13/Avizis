   106→  async start(taskInput: string, mode: RunMode, decisionMode: DecisionMode = 'manual'): Promise<RunSnapshot> {
   107→    const task = (taskInput || '').trim().slice(0, 500) || DEFAULT_TASK
   108→    const id = 'run_' + now().toString(36) + Math.random().toString(36).slice(2, 7)
   109→    const scenario = mode === 'demo' ? detectDemoScenario(task) : 'data'
   110→    const run: RunSnapshot = {
   111→      id,
   112→      task,
   113→      mode,
   114→      decisionMode,
   115→      status: 'planning',
   116→      createdAt: now(),
   117→      progress: 0,
   118→      nodes: [],
   119→      version: 0,
   120→      versions: [],
   121→      events: [],
   122→      logs: [],
   123→    }
   124→    const rt: Runtime = {
   125→      snapshot: run,
   126→      stopped: false,
   127→      paused: false,
   128→      loopActive: false,
   129→      dynamicAdded: 0,
   130→      demoAdaptDone: new Set(),
   131→      scenario,
   132→      demoInsertionsDone: new Set(),
   133→      emergentAdded: 0,
   134→      decisionMode,
   135→      pendingDecision: null,
   136→    }
   137→    this.runs.set(id, rt)
   138→
   139→    this.emit('run:created', { runId: id, task, mode, decisionMode, createdAt: run.createdAt })
   140→    this.emitEvent(rt, { type: 'decision:mode', ts: now(), mode: decisionMode })
   141→    this.pushStatus(rt, 'planning', 0)
   142→    this.log(rt, 'run', `Запуск агента · ${mode === 'demo' ? 'демо-режим (симуляция)' : 'LLM через z-ai-web-dev-sdk'} · задача: «${trunc(task, 90)}»`)
   143→
   144→    if (mode === 'llm') {
   145→      this.log(rt, 'plan', 'Планировщик LLM анализирует задачу…')
   146→      try {
   147→        await getZai()
   148→      } catch (e: any) {
   149→        this.log(rt, 'run', `Ошибка инициализации SDK: ${e?.message || e}`)
   150→        this.pushStatus(rt, 'error', 0)
   151→        store.save(run)
   152→        return run
   153→      }
   154→    } else {
   155→      this.log(rt, 'plan', 'Планировщик строит граф задачи…')
   156→      await sleep(900)
   157→    }
   158→
   159→    let nodes: GraphNode[]
   160→    if (mode === 'llm') {
   161→      nodes = await this.planWithLLM(task, rt)
   162→    } else if (scenario === 'writing') {
   163→      nodes = demoWritingPlan()
   164→      this.log(rt, 'plan', 'Демо-сценарий «Напиши пост»: размытый запрос — агент будет уточнять вектор по ходу выполнения')
   165→    } else {
   166→      nodes = demoPlan()
   167→    }
   168→
   169→    this.emitEvent(rt, { type: 'graph:init', ts: now(), nodes })
   170→    run.nodes = nodes.map((n) => ({ ...n }))
   171→    run.version = 1
   172→    run.versions.push({ version: 1, createdAt: now(), description: 'Исходный план агента', nodeIds: nodes.map((n) => n.id) })
   173→    this.log(rt, 'plan', `План готов: ${nodes.length} узлов · версия графа v1`)
   174→
   175→    // стартовая самооценка «понимания задачи»: насколько узкий вектор решения
   176→    const initialCertainty = mode === 'llm' ? (rt.snapshot.certainty ?? 0.4) : DEMO_INITIAL_CERTAINTY[scenario]
   177→    const initialNote = mode === 'llm' ? undefined : DEMO_INITIAL_CERTAINTY_NOTE[scenario]
   178→    this.emitCertainty(rt, initialCertainty, initialNote)
   179→
   180→    this.pushStatus(rt, 'running', 0)
   181→    this.log(rt, 'run', 'Выполнение графа началось')
   182→    store.save(run)
   183→
   184→    this.runLoop(rt).catch((err: any) => {
   185→      this.log(rt, 'run', `Внутренняя ошибка цикла выполнения: ${err?.message || err}`)
   186→      this.pushStatus(rt, 'error', run.progress)
   187→    })
   188→    return run
   189→  }

   191→  private async planWithLLM(task: string, rt: Runtime): Promise<GraphNode[]> {
   192→    const user = `Задача пользователя: «${task}»
   193→
   194→Сначала ОЦЕНИ: если запрос размыт (не заданы формат, объём, тон, аудитория, скоуп или подход) — начни понимание с низкой оценки certainty и включи развилки-уточнения. Если задача про разработку — уточняй продуктовые и архитектурные вопросы; если про тексты — tone of voice и формат; если про данные/анализ — стратегию проверки.
   195→
   196→Составь план из 4–7 шагов с ОДНОЙ-ДВУМЯ развилками (узлами-решениями): точками, где выбор одного или нескольких вариантов меняет дальнейшие шаги и итоговый результат. Формат ответа — строго один JSON-объект без каких-либо дополнений:
   197→
   198→{"certainty":0.3,"nodes":[
   199→  {"id":"n1","title":"краткое название шага","description":"что делает шаг, 1-2 предложения","dependsOn":[]},
   200→  {"id":"dec1","title":"Решение: ...","description":"суть развилки","dependsOn":["n1"],

   202→     {"id":"a","label":"название варианта","rationale":"обоснование: плюсы, минусы, цена выбора","steps":[
   203→       {"id":"ba1","title":"шаг ветки A","description":"..."}
   204→     ]},
   205→     {"id":"b","label":"название варианта","rationale":"...","steps":[
   206→       {"id":"bb1","title":"шаг ветки B","description":"..."}
   207→     ]}

   215→- Развилок не больше двух; вариантов 2–4; шагов в каждой ветке 1–3. Первый шаг ветки выполняется вместо шагов других веток.

   235→  /** Нормализация плана LLM: цепочки + узел-развилка с ветками вариантов + certainty */
   236→  private sanitizePlan(json: any, rt?: Runtime): GraphNode[] | null {
   237→    // стартовая самооценка «понимания задачи» от планировщика
   238→    if (rt) {
   239→      const c = Number(json?.certainty)
   240→      if (Number.isFinite(c)) rt.snapshot.certainty = Math.min(1, Math.max(0, c))
   241→    }
   242→    const rawNodes = Array.isArray(json?.nodes) ? json.nodes.slice(0, 12) : null
   243→    if (!rawNodes || rawNodes.length === 0) return null
   244→
   245→    const used = new Set<string>()
   246→    const cleanId = (v: any, fallback: string): string => {
   247→      let s = String(v ?? '').trim().toLowerCase().replace(/[^a-z0-9]/g, '').slice(0, 12)
   248→      if (!s) s = fallback
   249→      while (used.has(s)) s = (s.slice(0, 9) || 'n') + 'x' + Math.floor(Math.random() * 900 + 100)
   250→      used.add(s)
   251→      return s
   252→    }
   253→
   254→    const declared = new Set<string>()
   255→    const cleanDeps = (raw: any, exclude: Set<string>): string[] => {
   256→      const arr: any[] = Array.isArray(raw?.dependsOn) ? raw.dependsOn : []
   257→      const deps = arr
   258→        .map((d: any) => String(d ?? '').trim().toLowerCase().replace(/[^a-z0-9]/g, ''))
   259→        .filter((d: string) => d && declared.has(d) && !exclude.has(d))
   260→      return [...new Set(deps)].slice(0, 5)
   261→    }
   262→
   263→    const nodes: GraphNode[] = []
   264→    rawNodes.forEach((raw: any, i: number) => {
   265→      const id = cleanId(raw?.id, `n${i + 1}`)
   266→      const decisionRaw = raw?.decision
   267→      const optionsRaw = Array.isArray(decisionRaw?.options) ? decisionRaw.options.filter((o: any) => o).slice(0, 4) : []
   268→      const hasDecision = optionsRaw.length >= 2 && !!decisionRaw?.question
   269→
   270→      if (hasDecision) {
   271→        const branchIds = new Set<string>()
   272→        const selection: 'single' | 'mixed' = decisionRaw?.selection === 'mixed' ? 'mixed' : 'single'
   273→        const domain: DecisionDomain | undefined = DECISION_DOMAINS.includes(decisionRaw?.domain)
   274→          ? decisionRaw.domain
   275→          : undefined
   276→        const certaintyAfter = Number(decisionRaw?.certaintyAfter)
   277→        const options: DecisionOption[] = []
   278→        optionsRaw.forEach((ro: any, j: number) => {
   279→          const optId = ['a', 'b', 'c', 'd'][j]
   280→          const stepsRaw = Array.isArray(ro?.steps) ? ro.steps.filter((s: any) => s).slice(0, 3) : []
   281→          options.push({
   282→            id: optId,
   283→            label: String(ro?.label || `Вариант ${optId}`).slice(0, 60),
   284→            rationale: String(ro?.rationale || '').slice(0, 800),
   285→            nextSteps: stepsRaw.map((s: any) => String(s?.title || 'шаг ветки').slice(0, 60)),
   286→          })
   287→          // шаги ветки: цепочка от развилки
   288→          let prev = id
   289→          stepsRaw.forEach((rs: any, k: number) => {
   290→            const bid = cleanId(rs?.id, `${id}${optId}${k + 1}`)
   291→            branchIds.add(bid)
   292→            declared.add(bid)
   293→            nodes.push({
   294→              id: bid,
   295→              title: String(rs?.title || `Шаг ветки ${optId}${k + 1}`).slice(0, 60),
   296→              description: String(rs?.description || '').slice(0, 800),
   297→              status: 'pending',
   298→              dependsOn: [prev],
   299→              dynamic: false,
   300→              runs: 0,
   301→              branchOf: { decisionId: id, optionId: optId },
   302→            })
   303→            prev = bid
   304→          })
   305→        })
   306→        declared.add(id)
   307→        nodes.push({
   308→          id,
   309→          kind: 'decision',
   310→          title: String(raw?.title || 'Решение').slice(0, 60),
   311→          description: String(raw?.description || '').slice(0, 800),
   312→          status: 'pending',
   313→          dependsOn: cleanDeps(raw, branchIds),
   314→          dynamic: false,
   315→          runs: 0,
   316→          decision: {
   317→            question: String(decisionRaw.question).slice(0, 800),
   318→            domain,
   319→            selection,
   320→            certaintyAfter: Number.isFinite(certaintyAfter) ? Math.min(1, Math.max(0, certaintyAfter)) : undefined,
   321→            options,
   322→            status: 'pending',
   323→          },
   324→        })
   325→      } else {
   326→        declared.add(id)
   327→        nodes.push({
   328→          id,
   329→          title: String(raw?.title || `Шаг ${i + 1}`).slice(0, 60),
   330→          description: String(raw?.description || '').slice(0, 800),
   331→          status: 'pending',
   332→          dependsOn: cleanDeps(raw, new Set()),
   333→          dynamic: false,
   334→          runs: 0,
   335→        })
   336→      }
   337→    })
   338→    if (nodes.length < 2) return null
   339→    return nodes
   340→  }

   344→  private async runLoop(rt: Runtime) {
   345→    if (rt.loopActive) return
   346→    rt.loopActive = true
   347→    try {
   348→      while (true) {
   349→        const run = rt.snapshot
   350→        if (rt.stopped) {
   351→          this.finalizeStopped(rt)
   352→          return
   353→        }
   354→        const node = this.nextReadyNode(run)
   355→        if (!node) {
   356→          const errored = run.nodes.find((n) => n.status === 'error')
   357→          if (errored) {
   358→            this.log(rt, 'node-error', `Узел «${errored.title}» завершился ошибкой — выполнение остановлено. Доступно вмешательство: «Повторить узел».`)
   359→            this.pushStatus(rt, 'error', run.progress)
   360→            run.finishedAt = now()
   361→            store.save(run)
   362→            return
   363→          }
   364→          const pending = run.nodes.filter((n) => n.status === 'pending')
   365→          if (pending.length) {
   366→            this.log(rt, 'run', 'Обнаружена циклическая зависимость — оставшиеся узлы пропущены')
   367→            pending.forEach((n) => {
   368→              n.status = 'skipped'
   369→              this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...n } })
   370→            })
   371→          }
   372→          this.finalizeCompleted(rt)
   373→          return
   374→        }
   375→
   376→        await this.waitWhilePaused(rt)
   377→        if (rt.stopped) continue
   378→
   379→        node.input = this.buildInput(run, node)
   380→        node.status = 'running'
   381→        node.startedAt = now()
   382→        node.runs = (node.runs ?? 0) + 1
   383→        node.error = undefined
   384→        this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   385→        this.log(rt, 'node-start', `▶ «${node.title}» — выполнение`)
   386→
   387→        // ─── Узел-развилка: рассуждение → предложение вариантов → выбор → гашение веток ───
   388→        if (node.kind === 'decision' && node.decision) {
   389→          const decided = await this.handleDecisionNode(rt, node)
   390→          if (!decided) continue // остановлен во время решения — цикл завершит очистку
   391→          this.recalcProgress(rt)
   392→          this.pushStatus(rt, rt.paused ? 'paused' : 'running', run.progress)
   393→          store.save(run)
   394→          continue
   395→        }
   396→
   397→        try {
   398→          const output =
   399→            run.mode === 'demo' ? await this.demoExecute(node, rt) : await this.llmExecute(run, node)
   400→          node.status = 'done'
   401→          node.output = output
   402→          node.finishedAt = now()
   403→          node.durationMs = node.finishedAt - (node.startedAt ?? node.finishedAt)
   404→          this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   405→          this.log(rt, 'node-done', `✓ «${node.title}» — завершён за ${((node.durationMs ?? 0) / 1000).toFixed(1)} с`)
   406→        } catch (err: any) {
   407→          if (err instanceof StoppedError) {
   408→            node.status = 'pending'
   409→            node.startedAt = undefined
   410→            node.input = undefined
   411→            this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   412→            this.finalizeStopped(rt)
   413→            return
   414→          }
   415→          node.status = 'error'
   416→          node.error = err?.message || String(err)
   417→          node.finishedAt = now()
   418→          node.durationMs = node.finishedAt - (node.startedAt ?? node.finishedAt)
   419→          this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   420→          this.log(rt, 'node-error', `✗ «${node.title}» — ошибка: ${trunc(node.error ?? '', 140)}`)
   421→          continue
   422→        }
   423→
   424→        await this.maybeAdapt(rt, node)
   425→        this.recalcProgress(rt)
   426→        this.pushStatus(rt, rt.paused ? 'paused' : 'running', run.progress)
   427→        store.save(run)
   428→      }
   429→    } finally {
   430→      rt.loopActive = false
   431→    }
   432→  }

   434→  private nextReadyNode(run: RunSnapshot): GraphNode | undefined {
   435→    const byId = new Map(run.nodes.map((n) => [n.id, n]))
   436→    // «discarded» = ветка не реализована: зависимость от неё считается удовлетворённой
   437→    const ready = (n: GraphNode) =>
   438→      n.dependsOn.every((d) => {
   439→        const dep = byId.get(d)
   440→        return !dep || dep.status === 'done' || dep.status === 'skipped' || dep.status === 'discarded'
   441→      })
   442→    return run.nodes.find((n) => n.status === 'pending' && ready(n))
   443→  }

   454→      .join('\n')}`
   455→  }

   473→      if (rt.stopped) throw new StoppedError()
   474→      await sleep(120)

   626→  private insertNode(rt: Runtime, spec: { id: string; title: string; description: string; dependsOn: string[] }, reason: string) {
   627→    const run = rt.snapshot
   628→    const newNode: GraphNode = {
   629→      id: spec.id,
   630→      title: spec.title,
   631→      description: spec.description,
   632→      status: 'pending',
   633→      dependsOn: [...spec.dependsOn],
   634→      dynamic: true,
   635→      reason,
   636→      addedAtVersion: run.version + 1,
   637→      runs: 0,
   638→    }
   639→    const changed: GraphNode[] = [newNode]
   640→    const anchor = spec.dependsOn[0]
   641→    // Перепривязываем ещё не начатые узлы, зависящие от якоря, на новый узел
   642→    for (const n of run.nodes) {
   643→      if (n.status === 'pending' && n.dependsOn.includes(anchor)) {
   644→        n.dependsOn = [...new Set(n.dependsOn.map((d) => (d === anchor ? newNode.id : d)))]
   645→        changed.push({ ...n })
   646→      }
   647→    }
   648→    run.nodes.push(newNode)
   649→    run.version += 1
   650→    rt.dynamicAdded += 1
   651→    run.versions.push({ version: run.version, createdAt: now(), description: reason, nodeIds: run.nodes.map((n) => n.id) })
   652→    this.emitEvent(rt, { type: 'graph:adapt', ts: now(), reason, version: run.version, nodes: changed.map((n) => ({ ...n })) })
   653→    this.log(rt, 'adapt', `＋ Граф адаптирован (v${run.version}): добавлен узел «${newNode.title}». Причина: ${reason}`)
   654→    store.save(run)
   655→  }

   657→  /**
   658→   * Вставка эмерджентной развилки: агент проанализировал результаты шага и
   659→   * обнаружил неоднозначность, влияющую на итог. Создаёт узел-вопрос с ветками
   660→   * вариантов; ещё не начатые узлы, ждавшие якорь, переводятся на хвосты веток —
   661→   * итог корректно дождётся только реализованных веток.
   662→   */
   663→  private insertDecision(
   664→    rt: Runtime,
   665→    anchorId: string,
   666→    spec: {
   667→      id: string
   668→      title: string
   669→      description: string
   670→      question: string
   671→      domain?: string
   672→      selection: 'single' | 'mixed'
   673→      certaintyAfter?: number
   674→      options: Array<{ id: string; label: string; rationale: string; steps: Array<{ id?: string; title: string; description: string }> }>
   675→    },
   676→    reason: string
   677→  ) {
   678→    const run = rt.snapshot
   679→    const domain: DecisionDomain | undefined = DECISION_DOMAINS.includes(spec.domain as DecisionDomain)
   680→      ? (spec.domain as DecisionDomain)
   681→      : undefined
   682→    const version = run.version + 1
   683→
   684→    const decNode: GraphNode = {
   685→      id: spec.id,
   686→      kind: 'decision',
   687→      title: spec.title,
   688→      description: spec.description,
   689→      status: 'pending',
   690→      dependsOn: [anchorId],
   691→      dynamic: true,
   692→      reason,
   693→      addedAtVersion: version,
   694→      runs: 0,
   695→      decision: {
   696→        question: spec.question,
   697→        domain,
   698→        selection: spec.selection,
   699→        certaintyAfter: spec.certaintyAfter,
   700→        options: spec.options.map((o) => ({
   701→          id: o.id,
   702→          label: o.label,
   703→          rationale: o.rationale,
   704→          nextSteps: o.steps.map((s) => s.title),
   705→        })),
   706→        status: 'pending',
   707→      },
   708→    }
   709→    const changed: GraphNode[] = [decNode]
   710→    const tails: string[] = []
   711→
   712→    // ветки вариантов: цепочки от развилки
   713→    spec.options.forEach((o) => {
   714→      let prev = spec.id
   715→      o.steps.forEach((s, k) => {
   716→        const bid = s.id || `${spec.id}${o.id}${k + 1}`
   717→        const branchNode: GraphNode = {
   718→          id: bid,
   719→          title: s.title,
   720→          description: s.description,
   721→          status: 'pending',
   722→          dependsOn: [prev],
   723→          dynamic: true,
   724→          reason,
   725→          addedAtVersion: version,
   726→          runs: 0,
   727→          branchOf: { decisionId: spec.id, optionId: o.id },
   728→        }
   729→        run.nodes.push(branchNode)
   730→        changed.push({ ...branchNode })
   731→        prev = bid
   732→      })
   733→      if (prev === spec.id) return // у варианта нет шагов — ветка «прямое продолжение»
   734→      tails.push(prev)
   735→    })
   736→    run.nodes.push(decNode)
   737→
   738→    // перепривязка: ещё не начатые узлы, ждавшие якорь, ждут хвосты ВСЕХ веток
   739→    // (невыбранные ветки станут discarded — зависимость удовлетворится автоматически)
   740→    for (const n of run.nodes) {
   741→      if (n.id !== decNode.id && n.status === 'pending' && n.dependsOn.includes(anchorId)) {
   742→        n.dependsOn = [...new Set(n.dependsOn.flatMap((d) => (d === anchorId ? (tails.length ? tails : [anchorId]) : [d])))].slice(0, 12)
   743→        changed.push({ ...n })
   744→      }
   745→    }
   746→
   747→    run.version = version
   748→    rt.dynamicAdded += 1
   749→    rt.emergentAdded += 1
   750→    run.versions.push({ version, createdAt: now(), description: reason, nodeIds: run.nodes.map((n) => n.id) })
   751→    this.emitEvent(rt, { type: 'graph:adapt', ts: now(), reason, version, nodes: changed.map((n) => ({ ...n })) })
   752→    this.log(
   753→      rt,
   754→      'adapt',
   755→      `＋ Эмерджентная развилка (v${version}): «${spec.title}» — ${spec.options.length} варианта${
   756→        spec.selection === 'mixed' ? ' (смешанный выбор допустим)' : ''
   757→      }. Причина: ${reason}`
   758→    )
   759→    store.save(run)
   760→  }

   762→  /** Публикация самооценки «понимания задачи»: насколько узкий вектор решения */
   763→  private emitCertainty(rt: Runtime, certainty: number, note?: string) {
   764→    const c = Math.min(1, Math.max(0, Number.isFinite(certainty) ? certainty : 0))
   765→    rt.snapshot.certainty = c
   766→    this.emitEvent(rt, { type: 'run:certainty', ts: now(), certainty: c, note })
   767→    const pct = Math.round(c * 100)
   768→    this.log(rt, 'decision', `◎ Понимание задачи: ${pct}% — ${note || 'агент переоценил вектор решения'}`)
   769→  }

   773→  /** Жизненный цикл узла-развилки: рассуждение → предложение вариантов → выбор → активация ветки */
   774→  private async handleDecisionNode(rt: Runtime, node: GraphNode): Promise<boolean> {
   775→    const dec = node.decision!
   776→    this.log(rt, 'decision', `⎇ Развилка «${node.title}»: ${dec.question}`)
   777→    try {
   778→      await this.decisionThinkDelay(rt)
   779→    } catch {
   780→      return this.rollbackDecision(rt, node)
   781→    }
   782→    if (rt.stopped) return this.rollbackDecision(rt, node)
   783→
   784→    dec.status = 'offered'
   785→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   786→    this.emitEvent(rt, {
   787→      type: 'decision:offer',
   788→      ts: now(),
   789→      nodeId: node.id,
   790→      question: dec.question,
   791→      options: dec.options,
   792→    })
   793→
   794→    if (rt.decisionMode === 'manual') {
   795→      this.log(
   796→        rt,
   797→        'decision',
   798→        `⎇ Режим «ручной выбор»: агент предлагает ${dec.options.length} варианта с обоснованием и ожидает решения пользователя${
   799→          dec.selection === 'mixed' ? ' (допустимо смешанное решение из нескольких)' : ''
   800→        }`
   801→      )
   802→      const choice = await this.waitForDecision(rt, node)
   803→      if (!choice) return this.rollbackDecision(rt, node)
   804→      this.resolveDecision(rt, node, choice.optionIds, choice.by, choice.reasoning)
   805→    } else {
   806→      this.log(rt, 'decision', '⎇ Режим «авто»: агент взвешивает варианты и принимает решение самостоятельно')
   807→      const picked = await this.autoDecide(rt, node)
   808→      if (rt.stopped) return this.rollbackDecision(rt, node)
   809→      this.resolveDecision(rt, node, picked.optionIds, 'agent', picked.reasoning)
   810→    }
   811→    return true
   812→  }

   814→  /** Откат развилки (остановка во время решения): узел возвращается в очередь */
   815→  private rollbackDecision(rt: Runtime, node: GraphNode): boolean {
   816→    node.status = 'pending'
   817→    node.startedAt = undefined
   818→    node.input = undefined
   819→    if (node.decision) node.decision = { ...node.decision, status: 'pending' }
   820→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   821→    return false
   822→  }
   823→
   824→  /** Фаза рассуждения на развилке: пауза с уважением stop/pause */
   825→  private async decisionThinkDelay(rt: Runtime): Promise<void> {
   826→    const deadline = now() + (rt.snapshot.mode === 'llm' ? 700 : 1600)
   827→    while (now() < deadline) {
   828→      if (rt.stopped) throw new StoppedError()

   832→    }

   842→  /** Авто-режим: демо — скриптованный выбор; LLM — вызов модуля принятия решений */
   843→  private async autoDecide(rt: Runtime, node: GraphNode): Promise<{ optionIds: string[]; reasoning: string }> {
   844→    const dec = node.decision!
   845→    if (rt.snapshot.mode === 'demo') {
   846→      await sleep(1200)
   847→      const scripted = DEMO_WRITING_AUTO_PICKS[node.id]
   848→      if (scripted) return { optionIds: [...scripted.optionIds], reasoning: scripted.reasoning }
   849→      return { optionIds: [...DEMO_AUTO_PICK.optionIds], reasoning: DEMO_AUTO_PICK.reasoning }
   850→    }
   851→    const mixed = dec.selection === 'mixed'
   852→    const user = `Задача: «${rt.snapshot.task}»
   853→
   854→Развилка: ${dec.question}
   855→Тип выбора: ${mixed ? 'mixed — можно выбрать НЕСКОЛЬКО вариантов (смешанное решение)' : 'single — нужно выбрать ровно один вариант'}
   856→
   857→Варианты:
   858→${dec.options.map((o) => `- id «${o.id}» — ${o.label}: ${o.rationale}`).join('\n')}
   859→
   860→${node.input || ''}
   861→
   862→${mixed ? 'Выбери один или несколько вариантов (id из списка) и обоснуй состав.' : 'Выбери один вариант (id из списка) и обоснуй выбор.'} Ответ строго JSON:
   863→${mixed ? '{"optionIds":["<id1>","<id2>"],"reasoning":"обоснование в 1–2 предложениях"}' : '{"optionIds":["<id>"],"reasoning":"обоснование в 1–2 предложениях"}'}`
   864→    try {
   865→      const json = await chatJson(DECIDER_SYS, user, 45_000, 1)
   866→      const rawIds: any[] = Array.isArray(json?.optionIds)
   867→        ? json.optionIds
   868→        : Array.isArray(json?.options)
   869→          ? json.options
   870→          : json?.optionId
   871→            ? [json.optionId]
   872→            : []
   873→      const valid = rawIds
   874→        .map((v) => String(v ?? '').trim().toLowerCase())
   875→        .filter((v) => dec.options.some((o) => o.id === v))
   876→      const unique = [...new Set(valid)]
   877→      if (unique.length && json?.reasoning) {
   878→        const picked = mixed ? unique : unique.slice(0, 1)
   879→        return { optionIds: picked, reasoning: String(json.reasoning).slice(0, 800) }
   880→      }
   881→    } catch {
   882→      /* fallback ниже */
   883→    }
   884→    return { optionIds: [dec.options[0].id], reasoning: 'Обоснование LLM недоступно — выбран первый вариант по умолчанию' }
   885→  }

   887→  /** Применяет выбор (один или несколько вариантов): активирует ветки, гасит альтернативы, фиксирует аудит-трейл */
   888→  private resolveDecision(rt: Runtime, node: GraphNode, optionIds: string[], by: 'agent' | 'user', reasoning?: string) {
   889→    const run = rt.snapshot
   890→    const dec = node.decision!
   891→    const single = dec.selection !== 'mixed'
   892→    // валидация: только известные id; single — ровно один
   893→    let picked = optionIds.map((id) => id.trim().toLowerCase()).filter((id) => dec.options.some((o) => o.id === id))
   894→    picked = [...new Set(picked)]
   895→    if (!picked.length) picked = [dec.options[0].id]
   896→    if (single) picked = picked.slice(0, 1)
   897→    const opts = picked.map((id) => dec.options.find((o) => o.id === id)!)
   898→    const isMixedResult = picked.length > 1
   899→    const pickedLabel = opts.map((o) => `«${o.label}»`).join(isMixedResult ? ' + ' : '')
   900→
   901→    node.status = 'done'
   902→    node.finishedAt = now()
   903→    node.durationMs = node.finishedAt - (node.startedAt ?? node.finishedAt)
   904→    node.decision = { ...dec, status: 'resolved', chosen: picked, chosenBy: by, reasoning: reasoning?.trim() || undefined }
   905→    node.output =
   906→      by === 'agent'
   907→        ? `${isMixedResult ? `Смешанное решение агента: ${pickedLabel}.` : `Решение агента: вариант ${pickedLabel}.`} ${reasoning || ''}`.trim()
   908→        : `${isMixedResult ? `Смешанный выбор пользователя: ${pickedLabel}.` : `Выбор пользователя: вариант ${pickedLabel}.`} ${opts.map((o) => o.rationale).join(' ')}`
   909→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
   910→
   911→    // мгновенная визуализация последствий: нереализованные ветки гаснут
   912→    let discarded = 0
   913→    for (const n of run.nodes) {
   914→      if (n.branchOf?.decisionId === node.id && !picked.includes(n.branchOf.optionId) && n.status === 'pending') {
   915→        n.status = 'discarded'
   916→        discarded++
   917→        this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...n } })
   918→      }
   919→    }
   920→
   921→    this.emitEvent(rt, {
   922→      type: 'decision:resolved',
   923→      ts: now(),
   924→      nodeId: node.id,
   925→      optionIds: picked,
   926→      by,
   927→      reasoning: node.decision.reasoning,
   928→    })
   929→    this.log(
   930→      rt,
   931→      'decision',
   932→      `⎇ Решение принято (${by === 'agent' ? 'агентом' : 'пользователем'}): ${pickedLabel}${
   933→        isMixedResult ? ' — ветки активированы' : ' — ветка активирована'
   934→      }, узлов в нереализованных ветках: ${discarded}`
   935→    )
   936→
   937→    // закрытие вопроса сужает вектор понимания задачи
   938→    const after =
   939→      dec.certaintyAfter ?? Math.min(0.95, (rt.snapshot.certainty ?? 0.4) + 0.15)
   940→    this.emitCertainty(
   941→      rt,
   942→      Math.max(rt.snapshot.certainty ?? 0, after),
   943→      isMixedResult
   944→        ? `смешанное решение по вопросу «${trunc(node.title, 48)}»`
   945→        : `вопрос «${trunc(node.title, 48)}» закрыт`
   946→    )
   947→  }

   949→  /** Переключение режима решений на ходу; «авто» во время ожидания — агент решает немедленно */
   950→  setDecisionMode(runId: string, mode: DecisionMode): { ok: boolean; error?: string } {
   951→    const rt = this.runs.get(runId)
   952→    if (!rt) return { ok: false, error: 'Запуск не активен или уже выгружен' }
   953→    if (rt.decisionMode === mode) return { ok: true }
   954→    rt.decisionMode = mode
   955→    rt.snapshot.decisionMode = mode
   956→    this.emitEvent(rt, { type: 'decision:mode', ts: now(), mode })
   957→    this.log(
   958→      rt,
   959→      'control',
   960→      `⎇ Режим решений: ${mode === 'auto' ? '«авто» — агент решает сам' : '«ручной выбор» — агент предлагает варианты и ждёт'}`
   961→    )
   962→    store.save(rt.snapshot)
   963→    // если сейчас ожидается выбор, а включили «авто» — агент принимает решение вместо пользователя
   964→    if (mode === 'auto' && rt.pendingDecision) {
   965→      const pd = rt.pendingDecision
   966→      rt.pendingDecision = null
   967→      const node = rt.snapshot.nodes.find((n) => n.id === pd.nodeId)
   968→      if (node?.decision) {
   969→        this.log(rt, 'decision', '⎇ Ожидание выбора прервано: агент принимает решение вместо пользователя')
   970→        this.autoDecide(rt, node)
   971→          .then((picked) => pd.resolve({ optionIds: picked.optionIds, by: 'agent', reasoning: picked.reasoning }))
   972→          .catch(() => pd.resolve(null))
   973→      } else {
   974→        pd.resolve(null)
   975→      }
   976→    }
   977→    return { ok: true }
   978→  }

   980→  /** Выбор пользователя на развилке (из модалки): один или несколько вариантов */
   981→  resolveDecisionFromUser(runId: string, nodeId: string, optionIds: string[]): { ok: boolean; error?: string } {
   982→    const rt = this.runs.get(runId)
   983→    if (!rt) return { ok: false, error: 'Запуск не активен' }
   984→    const pd = rt.pendingDecision
   985→    if (!pd || pd.nodeId !== nodeId) return { ok: false, error: 'Решение не ожидается (уже принято или запуск завершён)' }
   986→    const node = rt.snapshot.nodes.find((n) => n.id === nodeId)
   987→    const dec = node?.decision
   988→    const ids = (optionIds || []).map((id) => String(id).trim().toLowerCase())
   989→    if (!dec || !ids.length || !ids.every((id) => dec.options.some((o) => o.id === id))) {
   990→      return { ok: false, error: 'Неизвестный вариант' }
   991→    }
   992→    rt.pendingDecision = null
   993→    pd.resolve({ optionIds: ids, by: 'user' })
   994→    return { ok: true }
   995→  }

   999→  control(runId: string, action: 'pause' | 'resume' | 'stop') {
  1000→    const rt = this.runs.get(runId)
  1001→    if (!rt) return
  1002→    if (action === 'pause') {
  1003→      if (!rt.paused && !rt.stopped && rt.snapshot.status === 'running') {
  1004→        rt.paused = true
  1005→        this.pushStatus(rt, 'paused', rt.snapshot.progress)
  1006→        this.log(rt, 'control', '⏸ Пауза: текущий узел завершится, новые не стартуют')
  1007→      }
  1008→    } else if (action === 'resume') {
  1009→      if (rt.paused) {
  1010→        rt.paused = false
  1011→        this.log(rt, 'control', '▶ Выполнение возобновлено')
  1012→        if (rt.snapshot.status === 'paused') {
  1013→          this.pushStatus(rt, 'running', rt.snapshot.progress)
  1014→        }
  1015→      }
  1016→    } else if (action === 'stop') {
  1017→      if (rt.stopped) return
  1018→      if (['completed', 'error', 'stopped'].includes(rt.snapshot.status)) return
  1019→      rt.stopped = true
  1020→      rt.paused = false
  1021→      // если ждём выбор пользователя на развилке — разблокируем ожидание
  1022→      if (rt.pendingDecision) {
  1023→        const pd = rt.pendingDecision
  1024→        rt.pendingDecision = null
  1025→        pd.resolve(null)
  1026→      }
  1027→      this.log(rt, 'control', '⏹ Остановка запрошена — применится после завершения текущего узла')
  1028→    }
  1029→  }

  1035→  private finalizeStopped(rt: Runtime) {
  1036→    const run = rt.snapshot
  1037→    let skipped = 0
  1038→    for (const n of run.nodes) {
  1039→      if (n.status === 'pending') {
  1040→        n.status = 'skipped'

  1069→  private recalcProgress(rt: Runtime) {
  1070→    const run = rt.snapshot
  1071→    // нереализованные ветки не считаются работой: прогресс по активным узлам
  1072→    const active = run.nodes.filter((n) => n.status !== 'discarded').length || 1
  1073→    const done = run.nodes.filter((n) => n.status === 'done').length
  1074→    run.progress = Math.round((done / active) * 100)
  1075→  }

  1077→  private downstreamOf(run: RunSnapshot, nodeId: string): GraphNode[] {
  1078→    const out: GraphNode[] = []
  1079→    const queue = [nodeId]
  1080→    const seen = new Set<string>()
  1081→    while (queue.length) {
  1082→      const cur = queue.shift()!
  1083→      for (const n of run.nodes) {
  1084→        if (n.dependsOn.includes(cur) && !seen.has(n.id)) {
  1085→          seen.add(n.id)
  1086→          out.push(n)
  1087→          queue.push(n.id)
  1088→        }
  1089→      }
  1090→    }
  1091→    return out
  1092→  }

  1094→  overrideNode(runId: string, nodeId: string, output: string): { ok: boolean; error?: string; resetCount?: number } {
  1095→    const rt = this.runs.get(runId)
  1096→    if (!rt) return { ok: false, error: 'Запуск не активен или уже выгружен' }
  1097→    const run = rt.snapshot
  1098→    const node = run.nodes.find((n) => n.id === nodeId && n.status === 'done')
  1099→    if (!node) return { ok: false, error: 'Узел не найден или ещё не завершён' }
  1100→    if (rt.stopped) return { ok: false, error: 'Запуск остановлен — переопределение недоступно' }
  1101→    const trimmed = (output || '').trim()
  1102→    if (!trimmed) return { ok: false, error: 'Результат не может быть пустым' }
  1103→
  1104→    node.output = trimmed.slice(0, 4000)
  1105→    node.overridden = true
  1106→    this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...node } })
  1107→    this.log(rt, 'intervention', `✎ Вмешательство: результат узла «${node.title}» переопределён пользователем`)
  1108→
  1109→    // нереализованные ветки не восстанавливаем — выбор на развилке остаётся в силе
  1110→    const down = this.downstreamOf(run, nodeId).filter((d) => d.status !== 'discarded')
  1111→    for (const d of down) {
  1112→      d.status = 'pending'
  1113→      d.output = undefined
  1114→      d.startedAt = undefined
  1115→      d.finishedAt = undefined
  1116→      d.durationMs = undefined
  1117→      d.error = undefined
  1118→      d.overridden = false
  1119→      this.emitEvent(rt, { type: 'node:update', ts: now(), node: { ...d } })
  1120→    }
  1121→    if (down.length) {
  1122→      this.log(rt, 'intervention', `↓ Пересчёт: ${down.length} зависимых узлов сброшены и будут выполнены заново`)
  1123→      this.recalcProgress(rt)
  1124→    }
  1125→    if (down.length && !rt.loopActive) {
  1126→      this.pushStatus(rt, 'running', run.progress)
  1127→      this.runLoop(rt).catch(() => {})
  1128→    }
  1129→    store.save(run)
  1130→    return { ok: true, resetCount: down.length }
  1131→  }
